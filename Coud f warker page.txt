function json(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      "Content-Type": "application/json; charset=utf-8",
      ...CORS_HEADERS
    }
  });
}

function generatePPId() {
  const chars = "23456789ABCDEFGHJKLMNPQRSTUVWXYZ"; // exclude 0, 1, I, O to avoid confusion
  let result = "";
  for (let i = 0; i < 8; i++) {
    result += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return result;
}

async function sendTelegram(chatId, text) {
  if (!chatId || !text) return;
  try {
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 2500);
    await fetch(`https://api.telegram.org/bot${TG_BOT_TOKEN}/sendMessage`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ chat_id: chatId, text, parse_mode: "HTML" }),
      signal: controller.signal
    });
    clearTimeout(timeout);
  } catch (e) {
    console.warn("Telegram alert failed:", e.message || e);
  }
}

        const note = body.note || body.title || "Coin Adjustment";
        const orderId = body.order_id || "";

        if (isNaN(coins) || coins === 0) return json({ ok: false, error: "Invalid coin amount" }, 400);

        const user = await db.prepare("SELECT pp_id, coins FROM users WHERE pp_id = ?").bind(uid).first();
        if (!user) return json({ ok: false, error: "User not found" }, 404);

        const currentCoins = Number(user.coins) || 0.0;
        const newBalance = Math.max(0, Math.round((currentCoins + coins + Number.EPSILON) * 100) / 100);
        const now = Date.now();

        await db.prepare("UPDATE users SET coins = ?, updated_at = ? WHERE pp_id = ?").bind(newBalance, now, uid).run();

        const txId = (coins > 0 ? "TX-CR-" : "TX-DR-") + now;
        await db.prepare(`
          INSERT INTO transactions (id, user_id, type, coins, balance_before, balance_after, title, details, created_at)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        `).bind(txId, uid, type, coins, currentCoins, newBalance, note, orderId, now).run();

        return json({
          ok: true,
          balance: newBalance,
          adjusted: coins
        });
      }

      // -------------------------------------------------------------
      // 5. TRANSACTIONS & HISTORY (10 per page pagination)
      // -------------------------------------------------------------
      if (path === "/api/transactions" && method === "GET") {
        const uid = (url.searchParams.get("user_id") || url.searchParams.get("pp_id") || "").toUpperCase();
        if (!uid) return json({ ok: false, error: "User ID required" }, 400);

        const page = Math.max(1, parseInt(url.searchParams.get("page") || "1", 10));
        const limit = Math.min(20, Math.max(1, parseInt(url.searchParams.get("limit") || "10", 10)));
        const offset = (page - 1) * limit;

        const totalRow = await db.prepare("SELECT COUNT(*) as total FROM transactions WHERE user_id = ?").bind(uid).first();
        const total = totalRow ? totalRow.total : 0;

        const rows = await db.prepare(`
          SELECT id, user_id, type, coins, balance_before, balance_after, title, details, created_at
          FROM transactions
          WHERE user_id = ?
          ORDER BY created_at DESC
          LIMIT ? OFFSET ?
        `).bind(uid, limit, offset).all();

        const list = (rows && rows.results) || [];
        const hasMore = (offset + list.length) < Math.min(200, total);

        const cappedTotal = Math.min(200, total);
        const totalPages = Math.max(1, Math.ceil(cappedTotal / limit));
        return json({
          ok: true,
          transactions: list,
          total: cappedTotal,
          pagination: {
            page,
            limit,
            total: cappedTotal,
            total_pages: totalPages,
            has_more: page < totalPages
          }
        });
      }

      // Legacy fallback endpoint for app compatibility
      if (path === "/api/history" && method === "GET") {
        const uid = (url.searchParams.get("user_id") || url.searchParams.get("uid") || url.searchParams.get("pp_id") || "").toUpperCase();
        const rows = await db.prepare(`
          SELECT id, user_id, type as kind, coins as amount, title as game, details as info, created_at
          FROM transactions WHERE user_id = ? ORDER BY created_at DESC LIMIT 50
        `).bind(uid).all();
        return json({ ok: true, history: (rows && rows.results) || [] });
      }

      // -------------------------------------------------------------
      // 6. PORTAL GAMES & QUESTS (Cloud Persistence for Render!)
      // -------------------------------------------------------------
      if (path === "/api/portal/games" && method === "GET") {
        const rows = await db.prepare(`
          SELECT game_id, title, package_name, icon_url, quests_json, cost_coins, is_active
          FROM portal_games
          WHERE is_active = 1
          ORDER BY sort_order ASC, title ASC
        `).all();

        const games = ((rows && rows.results) || []).map(g => {
          let quests = [];
          try { quests = JSON.parse(g.quests_json); } catch (e) { quests = []; }
          return {
            game_id: g.game_id,
            id: g.game_id,
            title: g.title,
            package_name: g.package_name,
            icon_url: g.icon_url,
            cost_coins: g.cost_coins,
            all_quests: quests
          };
        });

        return json({ ok: true, services: games, count: games.length });
      }

      // Admin: Upsert Game & its Quests into Cloud D1
      if (path === "/api/admin/portal/games" && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const gameId = (body.game_id || body.id || "").trim();
        const title = (body.title || "").trim();
        const pkg = (body.package_name || "").trim();
        const icon = body.icon_url || "";
        const cost = Number(body.cost_coins) || 2.0;
        const questsJson = typeof body.quests_json === "string" ? body.quests_json : JSON.stringify(body.all_quests || body.quests || []);

        if (!gameId || !title) return json({ ok: false, error: "game_id and title required" }, 400);

        const now = Date.now();
        await db.prepare(`
          INSERT INTO portal_games (game_id, title, package_name, icon_url, quests_json, cost_coins, is_active, updated_at)
          VALUES (?, ?, ?, ?, ?, ?, 1, ?)
          ON CONFLICT(game_id) DO UPDATE SET
            title = excluded.title,
            package_name = excluded.package_name,
            icon_url = excluded.icon_url,
            quests_json = excluded.quests_json,
            cost_coins = excluded.cost_coins,
            updated_at = excluded.updated_at
        `).bind(gameId, title, pkg, icon, questsJson, cost, now).run();

        return json({ ok: true, message: `Game ${title} saved permanently in Cloud D1` });

      // 7. ORDERS (Quest Unlocks)
      // -------------------------------------------------------------
      if (path === "/api/orders/create" && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const id = body.id || ("ORD-" + Date.now());
        const uid = (body.user_id || body.pp_id || "").toUpperCase();
        const email = body.user_email || body.email || "";
        const game = body.game || "Task";
        const logo = body.logo || "";
        const coins = Number(body.coins || body.amount) || 2.0;
        const qid = body.qid || body.sync_code || "";
        const now = Date.now();

        await db.prepare(`
          INSERT INTO orders (id, user_id, user_email, game, logo, coins, status, qid, sync_code, admin_msg, created_at)
          VALUES (?, ?, ?, ?, ?, ?, 'pending', ?, ?, 'সার্ভার কাজ শুরু করেছে...', ?)
        `).bind(id, uid, email, game, logo, coins, qid, qid, now).run();

        return json({ ok: true, order_id: id });
      }

      if (path === "/api/orders/update" && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const orderId = body.order_id || body.id || "";
        const status = body.status || "completed";
        const adminMsg = body.admin_msg || "";
        const timeElapsed = Number(body.time_elapsed) || 1;
        const now = Date.now();

        const order = await db.prepare("SELECT * FROM orders WHERE id = ?").bind(orderId).first();
        if (!order) return json({ ok: false, error: "Order not found" }, 404);

        if (status === "completed") {
          await db.prepare(`
            UPDATE orders SET status = 'completed', admin_msg = ?, time_elapsed = ?, completed_at = ? WHERE id = ?
          `).bind(adminMsg, timeElapsed, now, orderId).run();

          // Increment Leaderboard monthly point +1.0
          await db.prepare("UPDATE users SET monthly_points = monthly_points + 1.0 WHERE pp_id = ?").bind(order.user_id).run();

          sendTelegram(ORDER_LOG_CHANNEL, `⚡ <b>QUEST COMPLETED!</b> 🎉\n\n🎮 <b>Game:</b> ${order.game}\n👤 <b>User PP ID:</b> <code>${order.user_id}</code>\n🪙 <b>Coins:</b> ${order.coins} Coins\n⏱️ <b>Time:</b> ~${timeElapsed}s\n✅ <b>Status:</b> COMPLETED`);
        } else if (status === "refunded") {
          await db.prepare(`
            UPDATE orders SET status = 'refunded', admin_msg = ?, refunded_at = ? WHERE id = ?
          `).bind(adminMsg, now, orderId).run();

          // Auto-refund coins if not already refunded
          if (order.status !== "refunded" && order.coins > 0) {
            const user = await db.prepare("SELECT pp_id, coins FROM users WHERE pp_id = ?").bind(order.user_id).first();
            if (user) {
              const curCoins = Number(user.coins) || 0.0;
              const newCoins = Math.round((curCoins + order.coins + Number.EPSILON) * 100) / 100;
              await db.prepare("UPDATE users SET coins = ?, updated_at = ? WHERE pp_id = ?").bind(newCoins, now, order.user_id).run();
              const txId = "TX-REFUND-" + now;
              await db.prepare(`
                INSERT INTO transactions (id, user_id, type, coins, balance_before, balance_after, title, details, created_at)
                VALUES (?, ?, 'auto_refund', ?, ?, ?, ?, ?, ?)
              `).bind(txId, order.user_id, order.coins, curCoins, newCoins, `Refund: ${order.game} Quest`, orderId, now).run();
            }
          }

          sendTelegram(ORDER_LOG_CHANNEL, `↩️ <b>QUEST AUTO-REFUNDED!</b> 🪙\n\n🎮 <b>Game:</b> ${order.game}\n👤 <b>User PP ID:</b> <code>${order.user_id}</code>\n🪙 <b>Coins Back:</b> +${order.coins} Coins\n↩️ <b>Status:</b> 100% AUTO-REFUNDED`);
        }

        return json({ ok: true });
      }

      if (path === "/api/orders" && method === "GET") {
        const uid = (url.searchParams.get("user_id") || url.searchParams.get("pp_id") || "").toUpperCase();
        const rows = await db.prepare("SELECT * FROM orders WHERE user_id = ? ORDER BY created_at DESC LIMIT 50").bind(uid).all();
        return json({ ok: true, orders: (rows && rows.results) || [] });
      }

      if (path === "/api/admin/orders" && method === "GET") {
        const rows = await db.prepare("SELECT * FROM orders ORDER BY created_at DESC LIMIT 100").all();
        return json({ ok: true, orders: (rows && rows.results) || [] });
      }

      // -------------------------------------------------------------
      // 8. DEPOSITS (Top-up PP Coins)
      // -------------------------------------------------------------
      if ((path === "/api/deposits/create" || path === "/api/deposits") && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const uid = (body.user_id || body.pp_id || "").toUpperCase();
        const methodPay = body.method || "bKash";
        const taka = Number(body.amount_taka !== undefined ? body.amount_taka : body.amount) || 0.0;
        const coins = Number(body.coins) || 0.0;
        const trxId = (body.trx_id || "").trim().toUpperCase();

        if (!uid || taka <= 0 || coins <= 0 || !trxId) {
          return json({ ok: false, error: "সঠিক পেমেন্ট তথ্য দিন" }, 400);
        }

        const existingTrx = await db.prepare("SELECT id FROM deposits WHERE trx_id = ?").bind(trxId).first();
        if (existingTrx) {
          return json({ ok: false, error: "এই TrxID দিয়ে ইতিমধ্যে ডিপোজিট সাবমিট করা হয়েছে!" }, 400);
        }

        const user = await db.prepare("SELECT email FROM users WHERE pp_id = ?").bind(uid).first();
        const depId = "DEP-" + Date.now();
        const now = Date.now();

        await db.prepare(`
          INSERT INTO deposits (id, user_id, user_email, method, amount_taka, coins, trx_id, status, created_at)
          VALUES (?, ?, ?, ?, ?, ?, ?, 'pending', ?)
        `).bind(depId, uid, (user && user.email) || "", methodPay, taka, coins, trxId, now).run();

        sendTelegram(ADMIN_TG_ID, `💰 <b>NEW DEPOSIT REQUEST!</b>\n\n🆔 <b>PP ID:</b> <code>${uid}</code>\n📱 <b>Method:</b> ${methodPay}\n💵 <b>Amount:</b> ৳${taka} (${coins} PP Coins)\n🧾 <b>TrxID:</b> <code>${trxId}</code>`);

        return json({ ok: true, deposit_id: depId, message: "ডিপোজিট সফলভাবে সাবমিট হয়েছে। এডমিন ভেরিফাই করে কয়েন যুক্ত করে দিবে।" });
      }

      if (path === "/api/deposits" && method === "GET") {
        const uid = (url.searchParams.get("user_id") || url.searchParams.get("pp_id") || "").toUpperCase();
        const rows = await db.prepare("SELECT * FROM deposits WHERE user_id = ? ORDER BY created_at DESC LIMIT 50").bind(uid).all();
        return json({ ok: true, deposits: (rows && rows.results) || [] });
      }

      if (path === "/api/admin/deposits" && method === "GET") {
        const rows = await db.prepare("SELECT * FROM deposits ORDER BY created_at DESC LIMIT 100").all();
        return json({ ok: true, deposits: (rows && rows.results) || [] });
      }

      // Admin Deposit Approve / Reject
      if ((path === "/api/admin/deposits/review" || path === "/api/deposits/approve" || path === "/api/deposits/reject") && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const depId = body.deposit_id || body.id || body.dep_id || body.depId || "";
        let action = body.action;
        if (!action) {
          if (path.includes("approve")) action = "approve";
          else if (path.includes("reject")) action = "reject";
        }
        const adminMsg = body.admin_msg || (action === 'approve' ? "Approved by Admin" : "Rejected");
        const now = Date.now();

        const dep = await db.prepare("SELECT * FROM deposits WHERE id = ?").bind(depId).first();
        if (!dep) return json({ ok: false, error: "Deposit not found" }, 404);

        if (action === "approve") {
          // Strict duplicate prevention: if already approved, never credit again!
          if (dep.status === "approved") {
            return json({ ok: false, error: "⚠️ এই ডিপোজিটটি ইতিমধ্যেই অ্যাপ্রুভ করা হয়েছে!", alreadyApproved: true }, 400);
          }

          await db.prepare("UPDATE deposits SET status = 'approved', admin_msg = ?, reviewed_at = ? WHERE id = ?").bind(adminMsg, now, depId).run();

          // Credit coins to user with clean 2-decimal precision
          const user = await db.prepare("SELECT coins, telegram_chat_id, name, email FROM users WHERE pp_id = ?").bind(dep.user_id).first();
          if (user) {
            const currentCoins = Math.round(((Number(user.coins) || 0.0) + Number.EPSILON) * 100) / 100;
            const addCoins = Math.round(((Number(dep.coins) || 0.0) + Number.EPSILON) * 100) / 100;
            const newCoins = Math.round((currentCoins + addCoins + Number.EPSILON) * 100) / 100;
            await db.prepare("UPDATE users SET coins = ? WHERE pp_id = ?").bind(newCoins, dep.user_id).run();

            const txId = "TX-DEP-" + now;
            await db.prepare(`
              INSERT INTO transactions (id, user_id, type, coins, balance_before, balance_after, title, details, created_at)
              VALUES (?, ?, 'deposit', ?, ?, ?, 'Wallet Top-up', ?, ?)
            `).bind(txId, dep.user_id, addCoins, currentCoins, newCoins, `${dep.method} TrxID: ${dep.trx_id}`, now).run();

            // Send Telegram Notification on deposit approval
            try {
              const depTgMsg = `💎 <b>DEPOSIT APPROVED!</b> ✅\n\n` +
                `🆔 <b>User ID:</b> <code>${dep.user_id}</code>\n` +
                `💵 <b>Amount:</b> ${dep.amount_taka || 0} BDT\n` +
                `🪙 <b>Coins Credited:</b> +${addCoins} PP Coins\n` +
                `💰 <b>পূর্বের ব্যালেন্স:</b> ${currentCoins} Coins\n` +
                `✨ <b>নতুন ব্যালেন্স:</b> ${newCoins} Coins\n` +
                `💳 <b>Method:</b> ${dep.method || 'bKash'}\n` +
                `🔖 <b>TrxID:</b> <code>${dep.trx_id}</code>`;

              sendTelegram(ADMIN_TG_ID, depTgMsg);
              sendTelegram(ORDER_LOG_CHANNEL, depTgMsg);

              if (user.telegram_chat_id) {
                sendTelegram(user.telegram_chat_id,
                  `🎉 <b>আপনার ডিপোজিট সফলভাবে অ্যাপ্রুভ হয়েছে!</b> 💎\n\n` +
                  `💵 <b>টাকা:</b> ${dep.amount_taka || 0} BDT\n` +
                  `🪙 <b>যোগ হয়েছে:</b> +${addCoins} PP Coins\n` +
                  `💰 <b>পূর্বের ব্যালেন্স:</b> ${currentCoins} Coins\n` +
                  `✨ <b>বর্তমান ব্যালেন্স:</b> ${newCoins} Coins\n` +
                  `🔖 <b>TrxID:</b> <code>${dep.trx_id}</code>\n\n` +
                  `🚀 <i>এখনই আপনার পছন্দের গেমের কোয়েস্ট আনলক করুন!</i>`
                );
              }
            } catch (tgErr) {
              console.warn("Telegram deposit approval alert error:", tgErr);
            }
          }
        } else {
          await db.prepare("UPDATE deposits SET status = 'rejected', admin_msg = ?, reviewed_at = ? WHERE id = ?").bind(adminMsg, now, depId).run();
        }

        return json({ ok: true, status: action === 'approve' ? 'approved' : 'rejected' });
      }

      // -------------------------------------------------------------
      // 9. LEADERBOARD & REWARDS
      // -------------------------------------------------------------
      if (path === "/api/leaderboard" && method === "GET") {
        const rows = await db.prepare(`
          SELECT pp_id, name, avatar_url, monthly_points
          FROM users
          WHERE monthly_points > 0
          ORDER BY monthly_points DESC
          LIMIT 25
        `).all();

        return json({ ok: true, leaderboard: (rows && rows.results) || [] });
      }

      if (path === "/api/earn/daily-claim" && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const uid = (body.user_id || body.pp_id || "").toUpperCase();

        const lastTx = await db.prepare(`
          SELECT created_at FROM transactions
          WHERE user_id = ? AND type = 'daily_claim'
          ORDER BY created_at DESC LIMIT 1
        `).bind(uid).first();

        const now = Date.now();
        if (lastTx && (now - lastTx.created_at) < 86400000) {
          const waitHrs = Math.ceil((86400000 - (now - lastTx.created_at)) / 3600000);
          return json({ ok: false, error: `আজকের ক্লেইম ইতিমধ্যে নেওয়া হয়েছে! দয়া করে ${waitHrs} ঘণ্টা পর আবার চেষ্টা করুন।` }, 400);
        }

        const user = await db.prepare("SELECT coins FROM users WHERE pp_id = ?").bind(uid).first();
        if (!user) return json({ ok: false, error: "User not found" }, 404);

        const currentCoins = Number(user.coins) || 0.0;
        const newCoins = Math.round((currentCoins + 1.0 + Number.EPSILON) * 100) / 100;

        await db.prepare("UPDATE users SET coins = ? WHERE pp_id = ?").bind(newCoins, uid).run();

        const txId = "TX-DAILY-" + now;
        await db.prepare(`
          INSERT INTO transactions (id, user_id, type, coins, balance_before, balance_after, title, details, created_at)
          VALUES (?, ?, 'daily_claim', 1.0, ?, ?, 'Daily Check-in Bonus', 'Free +1 PP Coin', ?)
        `).bind(txId, uid, currentCoins, newCoins, now).run();

        return json({ ok: true, balance: newCoins, message: "🎉 অভিনন্দন! আপনি +১ PP Coin ডেইলি বোনাস পেয়েছেন!" });
      }

      // Promo Code Redeem
      if ((path === "/api/promo/redeem" || path === "/api/redeem") && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const uid = (body.user_id || body.pp_id || body.uid || "").toUpperCase();
        const code = (body.code || "").trim().toUpperCase();

        if (!uid || !code) {
          return json({ ok: false, error: "ইউজার আইডি ও প্রোমো কোড দিন" }, 400);
        }

        const user = await db.prepare("SELECT coins FROM users WHERE pp_id = ?").bind(uid).first();
        if (!user) {
          return json({ ok: false, error: "ইউজার একাউন্ট খুঁজে পাওয়া যায়নি" }, 404);
        }

        // Check if already redeemed by this user
        const alreadyRedeemed = await db.prepare("SELECT id FROM redemptions WHERE user_id = ? AND UPPER(code) = ?").bind(uid, code).first();
        if (alreadyRedeemed) {
          return json({ ok: false, error: "আপনি ইতিমধ্যে এই প্রোমো কোডটি ব্যবহার করে ফেলেছেন!" }, 400);
        }

        // Check promo_codes table first, or fall back to preset promo codes
        let promo = await db.prepare("SELECT * FROM promo_codes WHERE UPPER(code) = ? AND is_active = 1").bind(code).first();
        let rewardCoins = 0;

        if (promo) {
          if (promo.max_uses > 0 && promo.used_count >= promo.max_uses) {
            return json({ ok: false, error: "এই প্রোমো কোডটির ব্যবহারের সর্বোচ্চ সীমা শেষ হয়ে গেছে!" }, 400);
          }
          rewardCoins = Number(promo.coins) || 0;
        } else {
          const PRESET_CODES = {
            "WELCOME5": 5.0,
            "PPM2026": 10.0,
            "BONUS1": 1.0,
            "FREEPLAY": 2.0,
            "PPMOD": 3.0
          };
          if (PRESET_CODES[code]) {
            rewardCoins = PRESET_CODES[code];
          } else {
            return json({ ok: false, error: "অবৈধ প্রোমো কোড! কোডটি সঠিক নয় বা মেয়াদোত্তীর্ণ।" }, 400);
          }
        }

        const currentCoins = Number(user.coins) || 0.0;
        const newCoins = Math.round((currentCoins + rewardCoins + Number.EPSILON) * 100) / 100;
        const now = Date.now();

        await db.prepare("UPDATE users SET coins = ? WHERE pp_id = ?").bind(newCoins, uid).run();

        if (promo) {
          await db.prepare("UPDATE promo_codes SET used_count = used_count + 1 WHERE UPPER(code) = ?").bind(code).run();
        }

        const redId = "RED-" + now + "-" + Math.random().toString(36).substring(2, 6);
        await db.prepare("INSERT INTO redemptions (id, user_id, code, coins, created_at) VALUES (?, ?, ?, ?, ?)").bind(redId, uid, code, rewardCoins, now).run();

        const txId = "TX-PROMO-" + now;
        await db.prepare(`
          INSERT INTO transactions (id, user_id, type, coins, balance_before, balance_after, title, details, created_at)
          VALUES (?, ?, 'promo_redeem', ?, ?, ?, 'Promo Code Reward', ?, ?)
        `).bind(txId, uid, rewardCoins, currentCoins, newCoins, `Code: ${code}`, now).run();

        sendTelegram(ADMIN_TG_ID, `🎁 <b>PROMO CODE REDEEMED!</b>\n\n👤 <b>User:</b> <code>${uid}</code>\n🎟️ <b>Code:</b> <code>${code}</code>\n🪙 <b>Reward:</b> +${rewardCoins} Coins\n💰 <b>New Balance:</b> ${newCoins} Coins`);

        return json({
          ok: true,
          amount: rewardCoins,
          balance: newCoins,
          message: `🎉 অভিনন্দন! '${code}' প্রোমো কোডের মাধ্যমে +${rewardCoins} Coins পেয়েছেন!`
        });
      }

      // User Redemptions History
      if (path === "/api/promo/history" && method === "GET") {
        const uid = (url.searchParams.get("user_id") || url.searchParams.get("pp_id") || "").toUpperCase();
        if (!uid) return json({ ok: false, error: "user_id required" }, 400);
        const rows = await db.prepare("SELECT code, coins, created_at FROM redemptions WHERE user_id = ? ORDER BY created_at DESC LIMIT 20").bind(uid).all();
        return json({ ok: true, redemptions: (rows && rows.results) || [] });
      }

      // Earn Tasks List
      if (path === "/api/earn/tasks" && method === "GET") {
        const DEFAULT_TASKS = [
          {
            id: "daily_checkin",
            title: "দৈনিক বোনাস ক্লেইম",
            desc: "প্রতিদিন লগইন করে ফ্রি ১টি PP Coin উপহার নিন",
            reward: 1.0,
            resetType: "daily",
            icon: "🎁",
            actionType: "claim"
          },
          {
            id: "tg_channel",
            title: "Join Official Telegram",
            desc: "অফিশিয়াল টেলিগ্রাম চ্যানেলে জয়েন হয়ে আপডেট থাকুন",
            reward: 2.0,
            resetType: "once",
            icon: "📢",
            actionType: "link",
            url: "https://t.me/ppmodsofficial"
          },
          {
            id: "yt_channel",
            title: "Subscribe YouTube Channel",
            desc: "আমাদের ইউটিউব চ্যানেলে সাবস্ক্রাইব করুন",
            reward: 1.0,
            resetType: "once",
            icon: "▶️",
            actionType: "link",
            url: "https://youtube.com/@ppmods"
          },
          {
            id: "app_download",
            title: "Download PPM Server App",
            desc: "PPM কানেক্টর অ্যাপ v4.5 ডাউনলোড ও ইনস্টল করুন",
            reward: 5.0,
            resetType: "once",
            icon: "📲",
            actionType: "link",
            url: "https://t.me/ppmodsofficial"
          }
        ];
        return json({ ok: true, tasks: DEFAULT_TASKS });
      }

      // Claim Task Reward
      if (path === "/api/earn/claim-task" && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const uid = (body.user_id || body.pp_id || "").toUpperCase();
        const taskId = body.task_id || "";

        if (!uid || !taskId) return json({ ok: false, error: "user_id and task_id required" }, 400);

        const taskRewards = {
          "tg_channel": { coins: 2.0, title: "Telegram Channel Join Reward" },
          "yt_channel": { coins: 1.0, title: "YouTube Subscribe Reward" },
          "app_download": { coins: 5.0, title: "PPM Server App Install Reward" }
        };

        const taskInfo = taskRewards[taskId];
        if (!taskInfo) return json({ ok: false, error: "অবৈধ টাস্ক আইডি" }, 400);

        // Check if once task already completed
        const existingTx = await db.prepare("SELECT id FROM transactions WHERE user_id = ? AND type = 'task_reward' AND details = ?").bind(uid, taskId).first();
        if (existingTx) {
          return json({ ok: false, error: "আপনি ইতিমধ্যে এই টাস্কটি সম্পন্ন করে রিওয়ার্ড নিয়েছেন!" }, 400);
        }

        const user = await db.prepare("SELECT coins FROM users WHERE pp_id = ?").bind(uid).first();
        if (!user) return json({ ok: false, error: "User not found" }, 404);

        const currentCoins = Number(user.coins) || 0.0;
        const newCoins = Math.round((currentCoins + taskInfo.coins + Number.EPSILON) * 100) / 100;
        const now = Date.now();

        await db.prepare("UPDATE users SET coins = ? WHERE pp_id = ?").bind(newCoins, uid).run();

        const txId = "TX-TASK-" + now;
        await db.prepare(`
          INSERT INTO transactions (id, user_id, type, coins, balance_before, balance_after, title, details, created_at)
          VALUES (?, ?, 'task_reward', ?, ?, ?, ?, ?, ?)
        `).bind(txId, uid, taskInfo.coins, currentCoins, newCoins, taskInfo.title, taskId, now).run();

        return json({ ok: true, balance: newCoins, reward: taskInfo.coins, message: `🎉 অভিনন্দন! +${taskInfo.coins} Coins রিওয়ার্ড যুক্ত হয়েছে!` });
      }

      // -------------------------------------------------------------
      // 10. ADMIN MANAGEMENT
      // -------------------------------------------------------------
      // Public Config and Cache
      if ((path === "/api/cache/public-data" || path === "/api/config" || path === "/api/app/config") && method === "GET") {
        const row = await db.prepare("SELECT value FROM cloud_store WHERE key = 'config'").first();
        let cfg = {};
        if (row && row.value) {
          try { cfg = JSON.parse(row.value); } catch (e) { }
        }
        return json({ ok: true, config: cfg });
      }

      if ((path === "/api/cache/sync" || path === "/api/config/save" || path === "/api/admin/config") && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const cfg = body.config || body;
        const now = Date.now();
        await db.prepare(`
          INSERT INTO cloud_store (key, value, updated_at)
          VALUES ('config', ?, ?)
          ON CONFLICT(key) DO UPDATE SET value = excluded.value, updated_at = excluded.updated_at
        `).bind(JSON.stringify(cfg), now).run();
        return json({ ok: true, message: "Config saved to Cloudflare D1" });
      }

      // Cloud Store persistent sync for Render GPGS Quest Server
      if (path.startsWith("/api/cloud/") && method === "GET") {
        const key = path.replace("/api/cloud/", "");
        const row = await db.prepare("SELECT value FROM cloud_store WHERE key = ?").bind(key).first();
        if (!row) return json({ ok: true, data: [] });
        try {
          return json(JSON.parse(row.value));
        } catch (e) {
          return json({ ok: true, data: row.value });
        }
      }

      if (path.startsWith("/api/cloud/") && method === "POST") {
        const key = path.replace("/api/cloud/", "");
        const body = await request.json().catch(() => ({}));
        const valStr = JSON.stringify(body.data !== undefined ? body.data : body);
        const now = Date.now();
        await db.prepare(`
          INSERT INTO cloud_store (key, value, updated_at)
          VALUES (?, ?, ?)
          ON CONFLICT(key) DO UPDATE SET value = excluded.value, updated_at = excluded.updated_at
        `).bind(key, valStr, now).run();
        return json({ ok: true, message: `Synced ${key} to Cloudflare D1 successfully` });
      }

      // -------------------------------------------------------------
      // SYNC VAULT (Token & Pairing Code Management - Strict User Isolation)
      // -------------------------------------------------------------
      if (path === "/api/sync/save" && method === "POST") {
        const body = await request.json().catch(() => ({}));
        let email = (body.email || "").trim().toLowerCase();
        let token = (body.token || body.oauth_token || body.master_token || "").trim();
        let code = (body.code || "").trim();
        let country = (body.country || "GLOBAL").trim();
        let uid = (body.uid || body.user_id || "").trim().toUpperCase();
        let ownerEmail = (body.owner_email || "").trim().toLowerCase();
        let androidId = (body.android_id || "").trim();

        if (!code || !email || !token) {
          return json({ ok: false, error: "Missing required fields (code, email, token)" }, 400);
        }

        const now = Date.now();
        await db.prepare(`
          INSERT INTO pp_sync_vault (code, email, token, country, uid, owner_email, android_id, created_at, updated_at)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
          ON CONFLICT(code) DO UPDATE SET
            email = excluded.email,
            token = excluded.token,
            country = excluded.country,
            uid = excluded.uid,
            owner_email = excluded.owner_email,
            android_id = excluded.android_id,
            updated_at = excluded.updated_at
        `).bind(code, email, token, country, uid, ownerEmail, androidId, now, now).run();

        return json({ ok: true, code, message: "Account synced to cloud vault successfully" });
      }

      if (path === "/api/sync/accounts" && method === "GET") {
        const uid = (url.searchParams.get("uid") || url.searchParams.get("user_id") || "").trim().toUpperCase();
        const ownerEmail = (url.searchParams.get("owner_email") || url.searchParams.get("email") || "").trim().toLowerCase();

        // STRICT PRIVACY GUARD: Never return accounts if neither uid nor owner_email is provided!
        if (!uid && !ownerEmail) {
          return json({ ok: true, accounts: [] });
        }

        let rows;
        if (uid && ownerEmail) {
          rows = await db.prepare(`
            SELECT code, email, token, country, uid, owner_email, created_at, updated_at
            FROM pp_sync_vault
            WHERE (uid = ? OR owner_email = ? OR email = ?)
            ORDER BY updated_at DESC
          `).bind(uid, ownerEmail, ownerEmail).all();
        } else if (uid) {
          rows = await db.prepare(`
            SELECT code, email, token, country, uid, owner_email, created_at, updated_at
            FROM pp_sync_vault
            WHERE uid = ?
            ORDER BY updated_at DESC
          `).bind(uid).all();
        } else {
          rows = await db.prepare(`
            SELECT code, email, token, country, uid, owner_email, created_at, updated_at
            FROM pp_sync_vault
            WHERE (owner_email = ? OR email = ?)
            ORDER BY updated_at DESC
          `).bind(ownerEmail, ownerEmail).all();
        }

        return json({ ok: true, accounts: (rows && rows.results) || [] });
      }

      if (path === "/api/sync/get" && method === "GET") {
        const code = (url.searchParams.get("code") || "").trim();
        if (!code) return json({ ok: false, error: "Missing code" }, 400);

        const row = await db.prepare("SELECT * FROM pp_sync_vault WHERE code = ?").bind(code).first();
        if (!row) return json({ ok: false, error: "Code not found" }, 404);

        return json({
          ok: true,
          code: row.code,
          email: row.email,
          token: row.token,
          master_token: row.token,
          country: row.country,
          uid: row.uid,
          owner_email: row.owner_email,
          android_id: row.android_id || ""
        });
      }

      if (path === "/api/sync/delete" && (method === "POST" || method === "DELETE")) {
        const body = await request.json().catch(() => ({}));
        const code = (body.code || url.searchParams.get("code") || "").trim();
        const uid = (body.uid || body.user_id || url.searchParams.get("uid") || "").trim().toUpperCase();
        const email = (body.email || body.owner_email || url.searchParams.get("email") || "").trim().toLowerCase();

        if (!code) return json({ ok: false, error: "Missing code" }, 400);

        if (uid && email) {
          await db.prepare("DELETE FROM pp_sync_vault WHERE code = ? AND (uid = ? OR owner_email = ? OR email = ?)").bind(code, uid, email, email).run();
        } else if (uid) {
          await db.prepare("DELETE FROM pp_sync_vault WHERE code = ? AND uid = ?").bind(code, uid).run();
        } else if (email) {
          await db.prepare("DELETE FROM pp_sync_vault WHERE code = ? AND (owner_email = ? OR email = ?)").bind(code, email, email).run();
        } else {
          await db.prepare("DELETE FROM pp_sync_vault WHERE code = ?").bind(code).run();
        }

        return json({ ok: true, message: `Account ${code} deleted permanently` });
      }


      // Admin Stats Dashboard
      if (path === "/api/admin/stats" && method === "GET") {
        const uCount = (await db.prepare("SELECT COUNT(*) as c FROM users").first())?.c || 0;
        const oCount = (await db.prepare("SELECT COUNT(*) as c FROM orders").first())?.c || 0;
        const pDep = (await db.prepare("SELECT COUNT(*) as c FROM deposits WHERE status = 'pending'").first())?.c || 0;
        const totCoins = (await db.prepare("SELECT SUM(coins) as s FROM users").first())?.s || 0.0;
        const totalBal = Math.round((Number(totCoins) + Number.EPSILON) * 100) / 100;
        return json({
          ok: true,
          totalUsers: uCount,
          totalOrders: oCount,
          pendingDeposits: pDep,
          totalUserBalance: totalBal,
          stats: {
            totalUsers: uCount,
            totalOrders: oCount,
            pendingDeposits: pDep,
            totalCoins: totalBal
          }
        });
      }

      // Admin Users List & Pagination
      if ((path === "/api/admin/users" || path === "/api/users/list") && method === "GET") {
        const limit = Math.min(200, Math.max(1, parseInt(url.searchParams.get("limit") || "100", 10)));
        const offset = Math.max(0, parseInt(url.searchParams.get("offset") || "0", 10));
        const search = (url.searchParams.get("search") || "").trim().toLowerCase();

        let countRow;
        let rows;

        if (search) {
          const sPattern = `%${search}%`;
          countRow = await db.prepare("SELECT COUNT(*) as total, SUM(coins) as total_bal FROM users WHERE LOWER(email) LIKE ? OR UPPER(pp_id) LIKE ? OR LOWER(name) LIKE ?").bind(sPattern, sPattern, sPattern).first();
          rows = await db.prepare(`
            SELECT pp_id, email, password, name, avatar_url, coins, monthly_points, role, device_id, is_banned, ban_reason, created_at
            FROM users
            WHERE LOWER(email) LIKE ? OR UPPER(pp_id) LIKE ? OR LOWER(name) LIKE ?
            ORDER BY created_at DESC
            LIMIT ? OFFSET ?
          `).bind(sPattern, sPattern, sPattern, limit, offset).all();
        } else {
          countRow = await db.prepare("SELECT COUNT(*) as total, SUM(coins) as total_bal FROM users").first();
          rows = await db.prepare(`
            SELECT pp_id, email, password, name, avatar_url, coins, monthly_points, role, device_id, is_banned, ban_reason, created_at
            FROM users
            ORDER BY created_at DESC
            LIMIT ? OFFSET ?
          `).bind(limit, offset).all();
        }

        const rawList = (rows && rows.results) || [];
        const users = rawList.map(u => ({
          ...u,
          uid: u.pp_id,
          id: u.pp_id,
          balance: Number(u.coins) || 0,
          realBalance: Number(u.coins) || 0,
          bonusBalance: 0,
          banned: Boolean(u.is_banned),
          banReason: u.ban_reason || ""
        }));

        return json({
          ok: true,
          users,
          total: countRow ? countRow.total : 0,
          totalBalance: countRow ? (Number(countRow.total_bal) || 0) : 0
        });
      }

      // Admin Ban / Unban User
      if (path === "/api/user/ban" && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const uid = (body.user_id || body.uid || body.pp_id || "").trim().toUpperCase();
        const banned = body.banned ? 1 : 0;
        const reason = banned ? (body.banReason || body.reason || "Suspended by Admin") : "";
        if (!uid) return json({ ok: false, error: "user_id required" }, 400);

        await db.prepare("UPDATE users SET is_banned = ?, ban_reason = ? WHERE pp_id = ?").bind(banned, reason, uid).run();
        return json({ ok: true, message: `User ${uid} ban status updated to ${banned ? 'BANNED' : 'ACTIVE'}`, banned, ban_reason: reason });
      }

      // Admin Direct Set Balance
      if ((path === "/api/wallet/set-balance" || path === "/api/admin/set-balance") && method === "POST") {
        const body = await request.json().catch(() => ({}));
        const uid = (body.user_id || body.uid || body.pp_id || "").toUpperCase();
        const newBal = Number(body.balance !== undefined ? body.balance : body.coins) || 0.0;
        if (!uid) return json({ ok: false, error: "user_id required" }, 400);

        const curUser = await db.prepare("SELECT coins FROM users WHERE pp_id = ?").bind(uid).first();
        if (!curUser) return json({ ok: false, error: "User not found" }, 404);

        const curCoins = Number(curUser.coins) || 0.0;
        await db.prepare("UPDATE users SET coins = ? WHERE pp_id = ?").bind(newBal, uid).run();

        const now = Date.now();
        const txId = "TX-SETBAL-" + now;
        await db.prepare(`
          INSERT INTO transactions (id, user_id, type, coins, balance_before, balance_after, title, details, created_at)
          VALUES (?, ?, 'admin_adjust', ?, ?, ?, 'Admin Balance Set', 'Direct admin adjustment', ?)
        `).bind(txId, uid, (newBal - curCoins), curCoins, newBal, now).run();

        return json({ ok: true, balance: newBal });
      }

      if ((path === "/api/admin/user/delete" && method === "POST") || (path === "/api/admin/user" && method === "DELETE")) {
        const body = await request.json().catch(() => ({}));
        const uid = (body.user_id || body.pp_id || "").toUpperCase();
        if (!uid) return json({ ok: false, error: "user_id required" }, 400);

        await db.prepare("DELETE FROM users WHERE pp_id = ?").bind(uid).run();
        await db.prepare("DELETE FROM orders WHERE user_id = ?").bind(uid).run();
        await db.prepare("DELETE FROM deposits WHERE user_id = ?").bind(uid).run();
        await db.prepare("DELETE FROM transactions WHERE user_id = ?").bind(uid).run();

        return json({ ok: true, message: `User ${uid} deleted permanently from Cloudflare D1` });
      }

      return json({ ok: false, error: "Endpoint not found" }, 404);

    } catch (err) {
      console.error("API error:", err);
      return json({ ok: false, error: err.message || "Internal Worker Error" }, 500);
    }
  }
};
