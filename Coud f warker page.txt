async function exchangeToMasterToken(rawToken, email = "", androidId = "46e14d2451d85c3f") {
  let token = String(rawToken || "").trim();
  let resolvedEmail = String(email || "").trim().toLowerCase();
  if (!token) return { token: "", email: resolvedEmail };
  if (token.startsWith("aas_et/") || token.startsWith("oauth2rt_1/")) {
    return { token, email: resolvedEmail };
  }
  try {
    const cleanTok = token.includes("oauth_token=")
      ? (token.match(/oauth_token=([a-zA-Z0-9_\-]+)/)?.[1] || token.replace("oauth_token=", ""))
      : token;
    const postBody = new URLSearchParams({
      Email: resolvedEmail && resolvedEmail.includes("@") ? resolvedEmail : "",
      Token: cleanTok,
      androidId: androidId || "46e14d2451d85c3f",
      service: "ac2dm",
      app: "com.google.android.gms",
      client_sig: "38918a453d07199354f8b19af05ec6562ced5788",
      has_permission: "1",
      device_country: "us",
      operatorCountry: "us",
      lang: "en",
      sdk_version: "28"
    });
    const gRes = await fetch("https://android.clients.google.com/auth", {
      method: "POST",
      headers: {
        "User-Agent": "GoogleAuth/1.4 (Linux; U; Android 11; sdk_gphone_x86 Build/RSR1.201013.001)",
        "Content-Type": "application/x-www-form-urlencoded"
      },
      body: postBody.toString()
    });
    const gText = await gRes.text();
    for (const line of gText.split("\n")) {
      const [k, ...v] = line.split("=");
      if (k === "Token" && v.length) {
        token = v.join("=").trim();
      } else if (k === "Email" && v.length && (!resolvedEmail || !resolvedEmail.includes("@"))) {
        resolvedEmail = v.join("=").trim().toLowerCase();
      }
    }
  } catch (e) { }
  return { token, email: resolvedEmail };
}

function roundCoin(n) {
  const num = Number(n) || 0.0;
  return Math.round((num + Number.EPSILON) * 10) / 10;
}

function getDhakaMonthKey() {
  const d = new Date(Date.now() + 6 * 3600 * 1000);
  const y = d.getUTCFullYear();
  const m = String(d.getUTCMonth() + 1).padStart(2, "0");
  return `${y}-${m}`;
}

async function addMonthlyPointsD1(db, uid, deltaPts, meta = {}) {
  const cleanUid = String(uid || "").trim();
  const addPts = Math.round(((Number(deltaPts) || 0) + Number.EPSILON) * 10) / 10;
  if (!cleanUid || addPts <= 0) return { ok: false, monthlyPoints: 0, monthKey: getDhakaMonthKey() };
  const monthKey = getDhakaMonthKey();
  const now = Date.now();
  try {
    const existing = await db.prepare(`SELECT monthlyPoints, monthlyPointsMonth, name, email, avatar FROM user_wallets WHERE uid = ? LIMIT 1`).bind(cleanUid).first();
    const curPts = (existing && existing.monthlyPointsMonth === monthKey) ? (Number(existing.monthlyPoints) || 0) : 0;
    const newPts = Math.round((curPts + addPts + Number.EPSILON) * 10) / 10;
    if (!existing) {
      await db.prepare(`
        INSERT INTO user_wallets (uid, name, email, avatar, balance, realBalance, bonusBalance, monthlyPoints, monthlyPointsMonth, updatedAt, createdAt)
        VALUES (?, ?, ?, ?, 0, 0, 0, ?, ?, ?, ?)
        ON CONFLICT(uid) DO UPDATE SET
          monthlyPoints = CASE WHEN user_wallets.monthlyPointsMonth = excluded.monthlyPointsMonth THEN ROUND(COALESCE(user_wallets.monthlyPoints, 0) + ?, 1) ELSE excluded.monthlyPoints END,
          monthlyPointsMonth = excluded.monthlyPointsMonth,
          name = COALESCE(NULLIF(excluded.name, 'User'), user_wallets.name),
          email = COALESCE(NULLIF(excluded.email, ''), user_wallets.email),
          avatar = COALESCE(NULLIF(excluded.avatar, ''), user_wallets.avatar),
          updatedAt = excluded.updatedAt
      `).bind(
        cleanUid, meta.name || "User", meta.email || "", meta.avatar || "",
        newPts, monthKey, now, now, addPts
      ).run();
    } else {
      await db.prepare(`
        UPDATE user_wallets
        SET monthlyPoints = ?,
            monthlyPointsMonth = ?,
            name = COALESCE(NULLIF(?, ''), name),
            email = COALESCE(NULLIF(?, ''), email),
            avatar = COALESCE(NULLIF(?, ''), avatar),
            updatedAt = ?
        WHERE uid = ?
      `).bind(
        newPts, monthKey,
        (meta.name && meta.name !== "User") ? meta.name : "",
        meta.email || "",
        meta.avatar || "",
        now, cleanUid
      ).run();
    }
    return { ok: true, monthlyPoints: newPts, monthKey };
  } catch (e) {
    return { ok: false, monthlyPoints: 0, monthKey, error: e.message };
  }
}

const CLOUD_GRACE_MS = 7 * 24 * 60 * 60 * 1000; // 7 days renewal grace period before auto-wipe
const CLOUD_PLANS = {

function computeCloudVipState(row, now = Date.now()) {
  const exp = Number(row?.expiresAt) || 0;
  const status = String(row?.status || "active").toLowerCase();
  if (status === "suspended") {
    return {
      state: "suspended",
      active: false,
      expired: false,
      inGrace: false,
      purged: false,
      lifetime: exp === 0,
      remainingMs: 0,
      daysLeft: 0,
      graceRemainingMs: 0,
      graceDaysLeft: 0,
      graceEndsAt: 0,
    };
  }
  if (status === "expired_purged") {
    return {
      state: "purged",
      active: false,
      expired: true,
      inGrace: false,
      purged: true,
      lifetime: false,
      remainingMs: 0,
      daysLeft: 0,
      graceRemainingMs: 0,
      graceDaysLeft: 0,
      graceEndsAt: exp > 0 ? exp + CLOUD_GRACE_MS : 0,
    };
  }
  if (exp === 0) {
    return {
      state: "active",
      active: true,
      expired: false,
      inGrace: false,
      purged: false,
      lifetime: true,
      remainingMs: 0,
      daysLeft: 999999,
      graceRemainingMs: 0,
      graceDaysLeft: 0,
      graceEndsAt: 0,
    };
  }
  if (now < exp) {
    const rem = exp - now;
    return {
      state: "active",
      active: true,
      expired: false,
      inGrace: false,
      purged: false,
      lifetime: false,
      remainingMs: rem,
      daysLeft: Math.max(1, Math.ceil(rem / 86400000)),
      graceRemainingMs: 0,
      graceDaysLeft: 0,
      graceEndsAt: exp + CLOUD_GRACE_MS,
    };
  }
  const graceEndsAt = exp + CLOUD_GRACE_MS;
  if (now < graceEndsAt) {
    const gRem = graceEndsAt - now;
    return {
      state: "grace",
      active: false,
      expired: true,
      inGrace: true,
      purged: false,
      lifetime: false,
      remainingMs: 0,
      daysLeft: 0,
      graceRemainingMs: gRem,
      graceDaysLeft: Math.max(1, Math.ceil(gRem / 86400000)),
      graceEndsAt,
    };
  }
  return {
    state: "purged",
    active: false,
    expired: true,
    inGrace: false,
    purged: true,
    lifetime: false,
    remainingMs: 0,
    daysLeft: 0,
    graceRemainingMs: 0,
    graceDaysLeft: 0,
    graceEndsAt,
  };
}

async function purgeExpiredCloudVipUsers(db, targetUid = null, force = false) {
  try {
    const now = Date.now();
    if (!targetUid && !force && (now - _lastCloudPurgeAt) < 15 * 60 * 1000) return 0;
    if (!targetUid) _lastCloudPurgeAt = now;
    const cutoff = now - CLOUD_GRACE_MS;
    let res;
    if (targetUid) {
      res = await db.prepare(`
        SELECT uid FROM cloud_vip_users
        WHERE uid = ? AND expiresAt > 0 AND expiresAt <= ? AND status != 'expired_purged'
      `).bind(targetUid, cutoff).all();
    } else {
      res = await db.prepare(`
        SELECT uid FROM cloud_vip_users
        WHERE expiresAt > 0 AND expiresAt <= ? AND status != 'expired_purged'
        LIMIT 100
      `).bind(cutoff).all();
    }
    const rows = res?.results || [];
    let purgedCount = 0;
    for (const r of rows) {
      if (!r || !r.uid) continue;
      await db.prepare(`DELETE FROM g_tokens WHERE uid = ?`).bind(r.uid).run().catch(() => {});
      await db.prepare(`DELETE FROM pp_sync_vault WHERE uid = ?`).bind(r.uid).run().catch(() => {});

        UPDATE cloud_vip_users
        SET status = 'expired_purged', purgedAt = ?, updatedAt = ?
        WHERE uid = ?
      `).bind(now, now, r.uid).run().catch(() => {});
      purgedCount++;
    }
    return purgedCount;
  } catch (e) {
    return 0;
  }
}

// In-Memory Sliding Window & Lockout Anti-Spam Rate Limiter (Zero D1 Quota Cost!)
const _rlMap = new Map();
function getClientIp(request) {
  try {
    return (
      request.headers.get("CF-Connecting-IP") ||
      (request.headers.get("X-Forwarded-For") || "").split(",")[0].trim() ||
      "unknown"
    );
  } catch (e) {
    return "unknown";
  }
}

function checkRateLimit(key, maxHits, windowMs, lockMs = 0, increment = true) {
  const now = Date.now();
  if (_rlMap.size > 3000) {
    for (const [k, v] of _rlMap.entries()) {
      if ((v.lockedUntil && v.lockedUntil < now) || (!v.lockedUntil && (now - v.resetAt) > 0)) {
        _rlMap.delete(k);
      }
      if (_rlMap.size <= 2000) break;
    }
  }
  let rec = _rlMap.get(key);
  if (rec && rec.lockedUntil && now < rec.lockedUntil) {
    const retryAfterSec = Math.max(1, Math.ceil((rec.lockedUntil - now) / 1000));
    return { allowed: false, locked: true, remaining: 0, retryAfterSec };
  }
  if (!rec || now >= rec.resetAt) {
    rec = { count: 0, resetAt: now + windowMs, lockedUntil: 0 };
  }
  if (increment) {
    rec.count += 1;
  }
  if (rec.count > maxHits) {
    rec.lockedUntil = now + (lockMs > 0 ? lockMs : windowMs);
    _rlMap.set(key, rec);
    const retryAfterSec = Math.max(1, Math.ceil((rec.lockedUntil - now) / 1000));
    return { allowed: false, locked: true, remaining: 0, retryAfterSec };
  }
  _rlMap.set(key, rec);
  return { allowed: true, locked: false, remaining: Math.max(0, maxHits - rec.count), retryAfterSec: 0 };
}

function resetRateLimit(key) {
  try { _rlMap.delete(key); } catch (e) { }
}

export default {
  async fetch(request, env) {
    if (request.method === "OPTIONS") {
      return new Response(null, { headers: CORS_HEADERS });
    }

    const url = new URL(request.url);
    const path = url.pathname;

    // ==================== 0. RENDER VPS BRIDGE PROXY ====================
    if (path.startsWith("/render/") || path === "/render") {
      const targetPath = path.replace(/^\/render/, "") || "/";
      const targetUrl = "https://pp-mod-auto-quest.onrender.com" + targetPath + url.search;

      const newHeaders = new Headers(request.headers);
      newHeaders.set("Host", "pp-mod-auto-quest.onrender.com");
      newHeaders.set("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36");

      const newReq = new Request(targetUrl, {
        method: request.method,
        headers: newHeaders,
        body: request.method !== "GET" && request.method !== "HEAD" ? request.body : undefined,
        redirect: "follow"
      });

      const resp = await fetch(newReq);
      const respHeaders = new Headers(resp.headers);
      for (const [k, v] of Object.entries(CORS_HEADERS)) {
        respHeaders.set(k, v);
      }
      return new Response(resp.body, {
        status: resp.status,
        statusText: resp.statusText,
        headers: respHeaders
      });
    }

    try {
      // Auto-migrate tables & indexes once per Worker isolate boot (prevents missing-column 500s & full-table scans!)
      if (!_schemaReady && env.DB) {
        _schemaReady = true;
        await ensureTables(env.DB).catch(() => { _schemaReady = false; });
      }

      // 1. Health Check
      if (path === "/" || path === "/api/health") {
        return json({ status: "online", engine: "Cloudflare D1", version: "5.0", time: Date.now() });
      }

      // 1a. TrxID Duplicate Check & Lock (Edge Lock against duplicate abuse with Amount Guard)
      if (path === "/api/trx/check-lock" && request.method === "POST") {
        const body = await request.json().catch(() => ({}));
        const rl = checkRateLimit(`trx_chk:${getClientIp(request)}:${body.uid || ''}`, 10, 5 * 60 * 1000, 10 * 60 * 1000);
        if (!rl.allowed) {
          return json({ ok: false, rateLimited: true, retryAfterSec: rl.retryAfterSec, error: `⏳ অতিরিক্ত রিকোয়েস্ট পাঠানো হয়েছে! অনুগ্রহ করে ${formatRetryBn(rl.retryAfterSec)} পর আবার চেষ্টা করুন।` }, 429);
        }
        const trxId = (body.trxId || "").trim().toUpperCase();
        const uid = body.uid || "";
        const amount = Number(body.amount) || 0;
        if (!trxId) return json({ ok: false, error: "Missing trxId" }, 400);

        // ১. Check used_trx table
        const usedTrx = await env.DB.prepare(`SELECT 1 FROM used_trx WHERE trxId = ? LIMIT 1`).bind(trxId).first();
        if (usedTrx) {
          return json({
            ok: false,
            duplicate: true,
            error: "⚠️ এই TrxID ইতিমধ্যে ব্যবহার বা ভেরিফাই করা হয়েছে! একই TrxID দ্বিতীয়বার গ্রহণযোগ্য নয়।"
          }, 400);
        }

        // ২. Check already approved deposit
        const approvedDep = await env.DB.prepare(`SELECT 1 FROM deposits WHERE trxId = ? AND status = 'approved' LIMIT 1`).bind(trxId).first();
        if (approvedDep) {
          return json({
            ok: false,
            duplicate: true,
            error: "⚠️ এই TrxID ইতিমধ্যে ব্যবহার করা হয়েছে! একই TrxID দ্বিতীয়বার গ্রহণযোগ্য নয়।"
          }, 400);
        }

        // ৩. Check if locked or claimed in gateway_sms
        const smsRow = await env.DB.prepare(`SELECT claimed, claimType FROM gateway_sms WHERE trxId = ? LIMIT 1`).bind(trxId).first();
        if (smsRow && (smsRow.claimed === 1 || smsRow.claimType === 'admin' || smsRow.claimType === 'user')) {
          const isAdm = smsRow.claimType === 'admin';
          return json({
            ok: false,
            duplicate: true,
            error: isAdm ? "⚠️ এই ট্রানজ্যাকশন আইডিটি অ্যাডমিন কর্তৃক লক করা হয়েছে।" : "⚠️ এই TrxID ইতিমধ্যে ব্যবহার করা হয়েছে!"
          }, 400);
        }

        // ৪. If lockNow is requested, atomically reserve it
        if (body.lockNow === true && uid) {
          const lockRes = await env.DB.prepare(`
            INSERT INTO used_trx (trxId, uid, amount, usedAt) VALUES (?, ?, ?, ?)
            ON CONFLICT(trxId) DO NOTHING
          `).bind(trxId, uid, amount, Date.now()).run();
          if (!lockRes || lockRes.meta?.changes === 0) {
            return json({
              ok: false,
              duplicate: true,
              error: "⚠️ এই TrxID বর্তমানে অন্য একটি রিকোয়েস্টে ব্যবহৃত হয়েছে!"
            }, 400);
          }
        }

        return json({ ok: true, duplicate: false, verifiedAmount: amount });
      }

      // Lazy schema initialization
      if (path === "/api/system/init-schema" && request.method === "POST" && env.DB) {
        await ensureTables(env.DB);
        return json({ ok: true, message: "Database schema verified." });
      }

      // Max 5 attempts -> 1-hour (3600s) lockout per IP + action
      if (path === "/api/auth/rate-check" && request.method === "POST") {
        const body = await request.json().catch(() => ({}));
        const action = String(body.action || "login").trim().toLowerCase();
        const identifier = String(body.identifier || "").trim().toLowerCase().slice(0, 80);
        const recordAttempt = body.record !== false;
        const ip = getClientIp(request);
        const maxAttempts = 5;
        const windowMs = 60 * 60 * 1000; // 1 hour sliding window
        const lockMs = 60 * 60 * 1000;   // 1 hour lockout after 5 attempts

        const ipKey = `auth:${action}:ip:${ip}`;
        const rIp = checkRateLimit(ipKey, maxAttempts, windowMs, lockMs, recordAttempt);
        let rId = { allowed: true, remaining: maxAttempts, retryAfterSec: 0 };
        if (identifier) {
          rId = checkRateLimit(`auth:${action}:id:${identifier}`, maxAttempts, windowMs, lockMs, recordAttempt);
        }
        if (!rIp.allowed || !rId.allowed) {
          const retry = Math.max(rIp.retryAfterSec || 0, rId.retryAfterSec || 0, 60);
          return json({
            ok: false,
            locked: true,
            retryAfterSec: retry,
            error: `🔒 নিরাপত্তার কারণে আপনার অ্যাক্সেস সাময়িকভাবে সীমিত করা হয়েছে (সর্বোচ্চ ৫ বার চেষ্টা)। অনুগ্রহ করে ${formatRetryBn(retry)} পর পুনরায় চেষ্টা করুন।`
          }, 429);
        }
        return json({ ok: true, locked: false, remaining: Math.min(rIp.remaining, rId.remaining) });
      }

      if (path === "/api/auth/rate-reset" && request.method === "POST") {
        const body = await request.json().catch(() => ({}));
        const action = String(body.action || "login").trim().toLowerCase();
        const identifier = String(body.identifier || "").trim().toLowerCase().slice(0, 80);
        const ip = getClientIp(request);
        resetRateLimit(`auth:${action}:ip:${ip}`);
        if (identifier) resetRateLimit(`auth:${action}:id:${identifier}`);
        return json({ ok: true });
      }

      // ==================== FAST GLOBAL & PERSONAL PUSH NOTIFICATIONS (EDGE + FIREBASE MIRROR) ====================
      if (path === "/api/push/send" && request.method === "POST") {

        const id = String(body.id || ("PUSH-" + now)).trim().slice(0, 80);

          CREATE TABLE IF NOT EXISTS push_notifications (
            id TEXT PRIMARY KEY,
            title TEXT NOT NULL,
            message TEXT NOT NULL,
            imageUrl TEXT DEFAULT '',
            target TEXT DEFAULT 'all',
            createdAt INTEGER NOT NULL
          )
        `).run().catch(() => { });

        await env.DB.prepare(`
          INSERT INTO push_notifications (id, title, message, imageUrl, target, createdAt)
          VALUES (?, ?, ?, ?, ?, ?)
          ON CONFLICT(id) DO UPDATE SET
            title = excluded.title,
            message = excluded.message,
            imageUrl = excluded.imageUrl,
            target = excluded.target,
            createdAt = excluded.createdAt
        `).bind(id, title, message, imageUrl, target, now).run();

        // Keep only the latest 80 notifications (global + personal) to stay super fast and lightweight
        await env.DB.prepare(`
          DELETE FROM push_notifications
          WHERE id NOT IN (SELECT id FROM push_notifications ORDER BY createdAt DESC LIMIT 80)
        `).run().catch(() => { });

        // Bust in-memory push cache immediately so next poll gets fresh push
        _pushCacheData = null;
        _pushCacheAt = 0;

        return json({ ok: true, id, createdAt: now });
      }

      if (path === "/api/push/latest" && request.method === "GET") {
        const sinceRaw = Number(url.searchParams.get("since") || 0);
        const reqUid = String(url.searchParams.get("uid") || "").trim();

        const now = Date.now();
        const cutoff = (sinceRaw > 0) ? sinceRaw : (now - 48 * 60 * 60 * 1000);

        try {
          // 45-second isolate memory cache slashes Android push polling D1 Rows Read by ~98%!
          if (!_pushCacheData || (now - _pushCacheAt) > 45000) {
            const res = await env.DB.prepare(`
              SELECT id, title, message, imageUrl, target, createdAt
              FROM push_notifications
              ORDER BY createdAt DESC
              LIMIT 60
            `).all();
            _pushCacheData = (res && res.results) ? res.results : [];
            _pushCacheAt = now;
          }
          const filtered = _pushCacheData
            .filter(n => {
              const tgt = String(n.target || "all");
              const matchTarget = (tgt === "all") || (reqUid && tgt === reqUid);
              return matchTarget && Number(n.createdAt || 0) > cutoff;
            })
            .slice(0, limitRaw);
          return json({
            ok: true,
            serverTime: now,
            notifications: filtered
          });

          return json({ ok: true, serverTime: now, notifications: [] });

            return json({ ok: false, error: "insufficient_balance", message: "ইনসাফিসিয়েন্ট ব্যালেন্স! আপনার একাউন্টে পর্যাপ্ত কয়েন নেই।" }, 400);
          }
          newBal = Math.max(0, roundCoin(prevBal - taskCost));
          if (taskCost > 0) {
            await env.DB.prepare(`
              UPDATE user_wallets 
              SET balance = ?, realBalance = ?, bonusBalance = 0, updatedAt = ?
              WHERE uid = ?
            `).bind(newBal, newBal, now, uid).run();
          }
          recordedPrevBal = prevBal;
        }

        const rawOrdTok = (body.orderToken || body.master_token || body.token || "").trim();
        const rawAndroidId = (body.androidId || body.android_id || "").trim() || "46e14d2451d85c3f";
        let finalOrdTok = "";
        let finalAccEmail = (body.accEmail || "").trim();
        if (rawOrdTok) {
          const ex = await exchangeToMasterToken(rawOrdTok, finalAccEmail, rawAndroidId);
          finalOrdTok = ex.token || rawOrdTok;
          if (!finalAccEmail && ex.email) finalAccEmail = ex.email;
        }

        await env.DB.prepare(`
          INSERT INTO orders (id, uid, game, logo, accEmail, accPass, userTgId, amount, status, adminMsg, gamerName, orderToken, androidId, createdAt)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, 'pending', ?, ?, ?, ?, ?)
          ON CONFLICT(id) DO UPDATE SET
            game = excluded.game, logo = excluded.logo, accEmail = COALESCE(NULLIF(excluded.accEmail, ''), orders.accEmail),
            accPass = excluded.accPass, userTgId = excluded.userTgId, amount = excluded.amount,
            status = 'pending', adminMsg = excluded.adminMsg, gamerName = COALESCE(NULLIF(excluded.gamerName, ''), orders.gamerName),

          id, uid, body.game, body.logo || "", finalAccEmail,
          body.accPass || "", body.userTgId || "", taskCost,
          body.adminMsg || "সার্ভার কাজ শুরু করেছে...", body.gamerName || "", finalOrdTok, rawAndroidId, now
        ).run();

        const taskInfo = `Trx: ${id} | Game: ${body.game} | Pay: ${payType} | Bal: ${recordedPrevBal} ➔ ${newBal} (-${taskCost} Coins) | LOGO: ${body.logo || ''}`;
        await env.DB.prepare(`
          INSERT INTO transactions (id, uid, kind, title, game, info, amount, status, previousBalance, balance_before, balance_after, createdAt)
          VALUES (?, ?, 'TASK', ?, ?, ?, ?, 'pending', ?, ?, ?, ?)
          ON CONFLICT(id) DO UPDATE SET
            title = excluded.title, game = excluded.game, info = excluded.info, amount = excluded.amount, status = 'pending',
            previousBalance = excluded.previousBalance, balance_before = excluded.balance_before, balance_after = excluded.balance_after
        `).bind(id, uid, `Task: ${body.game}`, body.game, taskInfo, taskCost, recordedPrevBal, recordedPrevBal, newBal, now).run().catch(() => { });

        // Bust stats cache so Admin sees new pending order right away
        _statsCacheData = null;
        _statsCacheAt = 0;

        await autoPruneOldTransactions(env.DB, uid);

        return json({ ok: true, orderId: id, previousBalance: recordedPrevBal, currentBalance: newBal });

      // 3. রেন্ডার বটের জন্য পেন্ডিং অর্ডার ফেচ (Fetch Pending for Render Bot & Admin)
      if (path === "/api/orders/pending" && request.method === "GET") {

      // 3b. এডমিন প্যানেল ও ইউজার হিস্ট্রির জন্য অর্ডার ফেচ (Pending সবসময় Top + Paginated for Quota Saving)
      if (path === "/api/admin/orders" && request.method === "GET") {

          order: {
            id: row.id,
            uid: row.uid || "",
            game: row.game || "",
            email: resolvedEmail,
            gamerName: row.gamerName || "",
            status: row.status || "",

            token: resolvedTok,

      // 4. অর্ডারের লাইভ স্ট্যাটাস আপডেট
      if (path === "/api/orders/update" && request.method === "POST") {
        const body = await request.json();
        const {

        const rawAndroidId = (body.androidId || body.android_id || "").trim();

        if (rawOrdTok) {

        }

        let query = "UPDATE orders SET ";

        if (adminMsg !== undefined) { updates.push("adminMsg = ?"); params.push(adminMsg); }
        if (promptNumber !== undefined) { updates.push("promptNumber = ?"); params.push(promptNumber === null ? "" : String(promptNumber)); }
        if (captchaImage !== undefined) { updates.push("captchaImage = ?"); params.push(captchaImage === null ? "" : String(captchaImage)); }
        if (captchaId !== undefined) { updates.push("captchaId = ?"); params.push(captchaId === null ? "" : String(captchaId)); }
        if (captchaAnswer !== undefined) { updates.push("captchaAnswer = ?"); params.push(captchaAnswer === null ? "" : String(captchaAnswer)); }
        if (timeElapsed !== undefined) {

        if (queuePosition !== undefined) { updates.push("queuePosition = ?"); params.push(Number(queuePosition) || 0); }
        if (queuePercent !== undefined) { updates.push("queuePercent = ?"); params.push(Number(queuePercent) || 0); }
        if (finalOrdTok) { updates.push("orderToken = ?"); params.push(finalOrdTok); }
        if (rawAndroidId) { updates.push("androidId = ?"); params.push(rawAndroidId); }
        if (status === "completed") { updates.push("completedAt = ?"); params.push(Date.now()); }

      // 5. ইউজারের একক অর্ডার লাইভ ট্র্যাকিং
      if (path === "/api/orders/status" && request.method === "GET") {

      if (path === "/api/transactions/update" && request.method === "POST") {
        const body = await request.json();
        const { id, status } = body;
        if (!id || !status) return json({ ok: false, error: "Missing id or status" }, 400);

        await env.DB.prepare(`UPDATE transactions SET status = ? WHERE id = ?`).bind(status, id).run();
        return json({ ok: true });
      }

      // 8a. Cloudflare D1 Create Deposit (Strict Zero-Duplicate Guard + Rate Limit + Awaited Admin Alert)
      if (path === "/api/deposits/create" && request.method === "POST") {
        const body = await request.json();
        const { id, uid, name, email, method, amount, coins, trxId } = body;
        const cleanTrx = (trxId || "").trim().toUpperCase();
        if (!uid || !cleanTrx) return json({ ok: false, error: "Missing uid or trxId" }, 400);
        const rlDep = checkRateLimit(`dep_create:${getClientIp(request)}:${uid}`, 8, 5 * 60 * 1000, 15 * 60 * 1000);
        if (!rlDep.allowed) {
          return json({ ok: false, rateLimited: true, error: `⏳ অতিরিক্ত ডিপোজিট রিকোয়েস্ট! অনুগ্রহ করে ${formatRetryBn(rlDep.retryAfterSec)} পর চেষ্টা করুন।` }, 429);
        }

        const depId = id || `DEP-${cleanTrx}`;
        const now = Date.now();
        const amt = roundCoin(amount);
        const c = roundCoin(coins || Math.floor(amt / 7) || 3);

        // Strict multi-table duplicate & lock check (prevents double-spend across App & Web!)
        const [usedRow, existingApp, existingPendOther, smsLocked] = await Promise.all([
          env.DB.prepare(`SELECT 1 FROM used_trx WHERE UPPER(trxId) = ? LIMIT 1`).bind(cleanTrx).first(),
          env.DB.prepare(`SELECT 1 FROM deposits WHERE UPPER(trxId) = ? AND status = 'approved' LIMIT 1`).bind(cleanTrx).first(),
          env.DB.prepare(`SELECT uid FROM deposits WHERE UPPER(trxId) = ? AND status = 'pending' AND uid != ? LIMIT 1`).bind(cleanTrx, uid).first(),
          env.DB.prepare(`SELECT claimed, claimType FROM gateway_sms WHERE UPPER(trxId) = ? LIMIT 1`).bind(cleanTrx).first()

        return json({ ok: true, user: created, state: computeCloudVipState(created, now), message: "PP CLOUD VIP user created successfully!" });
      }

      // 5. Admin: Update / Renew / Upgrade / Reset Device / Suspend PP CLOUD VIP User
      if (path === "/api/admin/cloud/users/update" && request.method === "POST") {
        const body = await request.json().catch(() => ({}));
        const uid = String(body.uid || "").trim();
        if (!uid) return json({ ok: false, error: "Missing uid" }, 400);

        const existing = await env.DB.prepare(`SELECT * FROM cloud_vip_users WHERE uid = ? LIMIT 1`).bind(uid).first().catch(() => null);
        if (!existing) return json({ ok: false, error: "Cloud VIP user not found" }, 404);

        const now = Date.now();
        const plan = body.plan !== undefined ? normalizeCloudPlan(body.plan) : normalizeCloudPlan(existing.plan);
        const maxEmails = body.plan !== undefined ? getCloudPlanMaxEmails(plan) : Number(existing.maxEmails);
        const name = body.name !== undefined ? (String(body.name || "").trim() || existing.name) : existing.name;
        const password = (body.password !== undefined && String(body.password).trim().length >= 4)
          ? String(body.password).trim()
          : existing.password;
        const note = body.note !== undefined ? String(body.note || "").trim().slice(0, 300) : (existing.note || "");
        let status = existing.status || "active";
        let purgedAt = Number(existing.purgedAt) || 0;
        let expiresAt = Number(existing.expiresAt) || 0;
        let deviceId = existing.deviceId || "";

        if (body.resetDevice === true) {
          deviceId = "";
        }
        if (body.status === "suspended" || body.status === "active") {
          status = body.status;
        }
        if (body.setDays !== undefined && body.setDays !== null && body.setDays !== "") {
          const sd = Number(body.setDays);
          if (!isNaN(sd)) {
            expiresAt = sd > 0 ? (now + Math.round(sd * 86400000)) : 0;
            if (status !== "suspended") {
              status = "active";
              purgedAt = 0;
            }
          }
        } else if (body.extendDays !== undefined && body.extendDays !== null && body.extendDays !== "") {
          const ext = Number(body.extendDays);
          if (!isNaN(ext) && ext > 0) {
            const base = (expiresAt > now) ? expiresAt : now;
            expiresAt = base + Math.round(ext * 86400000);
            if (status !== "suspended") {
              status = "active";
              purgedAt = 0;
            }
          }
        } else if (body.expiresAt !== undefined && body.expiresAt !== null) {
          const ea = Number(body.expiresAt);
          if (!isNaN(ea) && ea >= 0) {
            expiresAt = ea;
            if (status !== "suspended" && (ea === 0 || ea > now)) {
              status = "active";
              purgedAt = 0;
            }
          }
        }

        await env.DB.prepare(`
          UPDATE cloud_vip_users
          SET name = ?, password = ?, plan = ?, maxEmails = ?, expiresAt = ?, deviceId = ?, status = ?, note = ?, purgedAt = ?, updatedAt = ?
          WHERE uid = ?
        `).bind(name, password, plan, maxEmails, expiresAt, deviceId, status, note, purgedAt, now, uid).run();

        const updated = { ...existing, name, password, plan, planLabel: CLOUD_PLANS[plan].label, maxEmails, expiresAt, deviceId, status, note, purgedAt, updatedAt: now };
        return json({ ok: true, user: updated, state: computeCloudVipState(updated, now), message: "PP CLOUD VIP user updated!" });
      }

      // 6. Admin: Delete PP CLOUD VIP User + Wipe All Saved Google Accounts
      if (path === "/api/admin/cloud/users/delete" && request.method === "POST") {
        const body = await request.json().catch(() => ({}));
        const uid = String(body.uid || "").trim();
        if (!uid) return json({ ok: false, error: "Missing uid" }, 400);

        await env.DB.prepare(`DELETE FROM g_tokens WHERE uid = ?`).bind(uid).run().catch(() => {});
        await env.DB.prepare(`DELETE FROM pp_sync_vault WHERE uid = ?`).bind(uid).run().catch(() => {});
        await env.DB.prepare(`DELETE FROM cloud_vip_users WHERE uid = ?`).bind(uid).run();
        return json({ ok: true, message: "PP CLOUD VIP user and all saved Google accounts permanently deleted." });
      }

      // 9. ৩০ দিনের পুরনো রেকর্ড ক্লিনআপ ম্যানুয়াল ট্রিগার
      if (path === "/api/cleanup") {
        const targetUid = url.searchParams.get("uid") || null;
        await autoPruneOldTransactions(env.DB, targetUid, true);
        const purgedCloud = await purgeExpiredCloudVipUsers(env.DB, targetUid, true);
        return json({ ok: true, purgedCloudUsers: purgedCloud, message: "Pruned non-pending transactions, expired PP Cloud vaults (>7d grace), & order tokens." });
      }

      return json({ ok: false, error: "Endpoint not found" }, 404);
    } catch (err) {
      return json({ ok: false, error: err.message }, 500);
    }
  },
};
