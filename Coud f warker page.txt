const CORS_HEADERS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type, Authorization",
  "Access-Control-Max-Age": "86400",
};

function json(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      "Content-Type": "application/json",
      ...CORS_HEADERS,
    },
  });
}

// 60-day auto-cleanup: keeps pending forever, purges records older than 60 days
async function autoPruneOldTransactions(db) {
  try {
    const sixtyDaysAgo = Date.now() - (60 * 24 * 60 * 60 * 1000);
    await db.prepare(`
      DELETE FROM transactions WHERE status != 'pending' AND createdAt < ?
    `).bind(sixtyDaysAgo).run();
    await db.prepare(`
      DELETE FROM orders WHERE status != 'pending' AND createdAt < ?
    `).bind(sixtyDaysAgo).run();
  } catch (e) {
    // ignore if tables not yet created
  }
}

export default {
  async fetch(request, env) {
    if (request.method === "OPTIONS") {
      return new Response(null, { headers: CORS_HEADERS });
    }

    const url = new URL(request.url);
    const path = url.pathname;

    // ==================== 0. RENDER VPS BRIDGE PROXY ====================
    if (path.startsWith("/render/") || path === "/render") {
      const targetPath = path.replace(/^\/render/, "") || "/";
      const targetUrl = "https://pp-mod-auto-quest.onrender.com" + targetPath + url.search;

      const newHeaders = new Headers(request.headers);
      newHeaders.set("Host", "pp-mod-auto-quest.onrender.com");
      newHeaders.set("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36");

      const newReq = new Request(targetUrl, {
        method: request.method,
        headers: newHeaders,
        body: request.method !== "GET" && request.method !== "HEAD" ? request.body : undefined,
        redirect: "follow"
      });

      const resp = await fetch(newReq);
      const respHeaders = new Headers(resp.headers);
      for (const [k, v] of Object.entries(CORS_HEADERS)) {
        respHeaders.set(k, v);
      }
      return new Response(resp.body, {
        status: resp.status,
        statusText: resp.statusText,
        headers: respHeaders
      });
    }

    try {
      // 1. Health Check
      if (path === "/" || path === "/api/health") {
        return json({ status: "online", engine: "Cloudflare D1", time: Date.now() });
      }

      // 1a. TrxID Duplicate Check & Lock (Edge Lock against duplicate abuse with Amount Guard)
      if (path === "/api/trx/check-lock" && request.method === "POST") {
        const body = await request.json().catch(() => ({}));
        const trxId = (body.trxId || "").trim().toUpperCase();
        const uid = body.uid || "";
        const amount = Number(body.amount) || 0;
        if (!trxId) return json({ ok: false, error: "Missing trxId" }, 400);

        // ১. Check used_trx table
        const usedTrx = await env.DB.prepare(`SELECT 1 FROM used_trx WHERE trxId = ? LIMIT 1`).bind(trxId).first();
        if (usedTrx) {
          return json({
            ok: false,
            duplicate: true,
            error: "⚠️ এই TrxID ইতিমধ্যে ব্যবহার বা ভেরিফাই করা হয়েছে! একই TrxID দ্বিতীয়বার গ্রহণযোগ্য নয়।"
          }, 400);
        }

        // ২. Check already approved deposit
        const approvedDep = await env.DB.prepare(`SELECT 1 FROM deposits WHERE trxId = ? AND status = 'approved' LIMIT 1`).bind(trxId).first();
        if (approvedDep) {
          return json({
            ok: false,
            duplicate: true,
            error: "⚠️ এই TrxID ইতিমধ্যে ব্যবহার করা হয়েছে! একই TrxID দ্বিতীয়বার গ্রহণযোগ্য নয়।"
          }, 400);
        }

        // ৩. Check if locked or claimed in gateway_sms
        const smsRow = await env.DB.prepare(`SELECT claimed, claimType FROM gateway_sms WHERE trxId = ? LIMIT 1`).bind(trxId).first();
        if (smsRow && (smsRow.claimed === 1 || smsRow.claimType === 'admin' || smsRow.claimType === 'user')) {
          const isAdm = smsRow.claimType === 'admin';
          return json({
            ok: false,
            duplicate: true,
            error: isAdm ? "⚠️ এই ট্রানজ্যাকশন আইডিটি অ্যাডমিন কর্তৃক লক করা হয়েছে।" : "⚠️ এই TrxID ইতিমধ্যে ব্যবহার করা হয়েছে!"
          }, 400);
        }

        // ৪. If lockNow is requested, atomically reserve it
        if (body.lockNow === true && uid) {
          const lockRes = await env.DB.prepare(`
            INSERT INTO used_trx (trxId, uid, amount, usedAt) VALUES (?, ?, ?, ?)
            ON CONFLICT(trxId) DO NOTHING
          `).bind(trxId, uid, amount, Date.now()).run();
          if (!lockRes || lockRes.meta?.changes === 0) {
            return json({
              ok: false,
              duplicate: true,
              error: "⚠️ এই TrxID বর্তমানে অন্য একটি রিকোয়েস্টে ব্যবহৃত হয়েছে!"
            }, 400);
          }
        }

        return json({ ok: true, duplicate: false, verifiedAmount: amount });
      }

      // Lazy schema initialization (Prevents Error 1102: Worker exceeded resource limits)
      if (path === "/api/system/init-schema" && request.method === "POST" && env.DB) {
        await ensureTables(env.DB);
        return json({ ok: true, message: "Database schema verified." });
      }

      // ==================== CLOUDFLARE D1 EDGE WALLET & DATA SHIFT ENGINE ====================

      // 1b. Fetch User Wallet (50 Lakh Daily Free Reads)
      if (path === "/api/wallet" && request.method === "GET") {
        const uid = url.searchParams.get("uid");
        if (!uid) return json({ ok: false, error: "Missing uid" }, 400);

        let row = await env.DB.prepare(`SELECT * FROM user_wallets WHERE uid = ?`).bind(uid).first();
        if (!row) {
          await env.DB.prepare(`
            INSERT INTO user_wallets (uid, balance, realBalance, bonusBalance, playPoints, pendingPoints, loyaltyStars, banned, banReason, updatedAt)
            VALUES (?, 0, 0, 0, 0, 0, 0, 0, '', ?)
            ON CONFLICT(uid) DO NOTHING
          `).bind(uid, Date.now()).run();
          row = { uid, balance: 0, realBalance: 0, bonusBalance: 0, playPoints: 0, pendingPoints: 0, loyaltyStars: 0, banned: 0, banReason: "" };
        }
        return json({ ok: true, wallet: row });
      }

      // 1c. Fast Task Payment (Instant deduction directly on D1 - zero firebase quota)
      if (path === "/api/wallet/pay-task" && request.method === "POST") {
        const body = await request.json();
        const { uid, feeCoins, isFree } = body;
        if (!uid) return json({ ok: false, error: "Missing uid" }, 400);

        let w = await env.DB.prepare(`SELECT * FROM user_wallets WHERE uid = ?`).bind(uid).first();
        if (!w) {
          w = { uid, balance: 0, realBalance: 0, bonusBalance: 0, loyaltyStars: 0 };
        }

        const fee = isFree ? 0 : Number(feeCoins || 3);
        const curReal = Number(w.realBalance || 0);
        const curBonus = Number(w.bonusBalance || 0);
        const totalBal = Number(w.balance || (curReal + curBonus));

        if (!isFree && totalBal < fee) {
          return json({ ok: false, error: "insufficient", balance: totalBal }, 400);
        }

        let deductReal = Math.min(curReal, fee);
        let deductBonus = Math.max(0, fee - deductReal);
        let newReal = Math.max(0, Math.round((curReal - deductReal + Number.EPSILON) * 100) / 100);
        let newBonus = Math.max(0, Math.round((curBonus - deductBonus + Number.EPSILON) * 100) / 100);
        let newTotal = Math.round((newReal + newBonus + Number.EPSILON) * 100) / 100;
        // পেমেন্টের সময় কোনো স্টার বাড়বে না — শুধুমাত্র টাস্ক COMPLETED হলেই স্টার বাড়বে!
        let newStars = Number(w.loyaltyStars || 0);

        await env.DB.prepare(`
          INSERT INTO user_wallets (uid, balance, realBalance, bonusBalance, loyaltyStars, updatedAt)
          VALUES (?, ?, ?, ?, ?, ?)
          ON CONFLICT(uid) DO UPDATE SET
            balance = excluded.balance,
            realBalance = excluded.realBalance,
            bonusBalance = excluded.bonusBalance,
            loyaltyStars = excluded.loyaltyStars,
            updatedAt = excluded.updatedAt
        `).bind(uid, newTotal, newReal, newBonus, newStars, Date.now()).run();

        return json({ ok: true, balance: newTotal, realBalance: newReal, bonusBalance: newBonus, loyaltyStars: newStars });
      }

      // 1d. Credit Wallet Balance & Loyalty Stars (Deposits, Refunds, Task Stars)
      if (path === "/api/wallet/credit" && request.method === "POST") {
        const body = await request.json();
        const { uid, coins, type, playPoints, pendingPoints, loyaltyStarsInc } = body;
        if (!uid) return json({ ok: false, error: "Missing uid" }, 400);

        let w = await env.DB.prepare(`SELECT * FROM user_wallets WHERE uid = ?`).bind(uid).first();
        if (!w) {
          w = { uid, balance: 0, realBalance: 0, bonusBalance: 0, playPoints: 0, pendingPoints: 0, loyaltyStars: 0 };
        }

        let curReal = Number(w.realBalance || 0);
        let curBonus = Number(w.bonusBalance || 0);
        let addCoins = Number(coins || 0);
        const prevBal = Number(w.balance || (curReal + curBonus));

        if (type === "bonus") {
          curBonus = Math.max(0, Math.round((curBonus + addCoins + Number.EPSILON) * 100) / 100);
        } else {
          curReal = Math.max(0, Math.round((curReal + addCoins + Number.EPSILON) * 100) / 100);
        }
        let newTotal = Math.round((curReal + curBonus + Number.EPSILON) * 100) / 100;

        let newPlayPoints = playPoints !== undefined ? Number(playPoints) : Number(w.playPoints || 0);
        let newPendingPoints = pendingPoints !== undefined ? Number(pendingPoints) : Number(w.pendingPoints || 0);

        // শুধুমাত্র টাস্ক সফলভাবে COMPLETED হলে স্টার বাড়বে (এবং ১০ম ফ্রি টাস্ক সম্পন্ন হলে ০ এ রিসেট হবে)
        let curStars = Number(w.loyaltyStars || 0);
        let newStars = body.resetLoyaltyStars ? 0 : (loyaltyStarsInc ? Math.min(10, curStars + Number(loyaltyStarsInc)) : curStars);

        await env.DB.prepare(`
          INSERT INTO user_wallets (uid, balance, realBalance, bonusBalance, playPoints, pendingPoints, loyaltyStars, updatedAt)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?)
          ON CONFLICT(uid) DO UPDATE SET
            balance = excluded.balance,
            realBalance = excluded.realBalance,
            bonusBalance = excluded.bonusBalance,
            playPoints = excluded.playPoints,
            pendingPoints = excluded.pendingPoints,
            loyaltyStars = excluded.loyaltyStars,
            updatedAt = excluded.updatedAt
        `).bind(uid, newTotal, curReal, curBonus, newPlayPoints, newPendingPoints, newStars, Date.now()).run();

        // অডিট ট্রেইল: কয়েন যোগ বা কাটা হলে সাথে সাথে ট্রানজ্যাকশন হিস্ট্রি টেবিল রেকর্ড করা
        if (addCoins !== 0 && body.skipTxLog !== true) {
          const txId = body.txId || `TX-${Date.now()}-${Math.random().toString(36).substr(2, 5).toUpperCase()}`;
          const isAdd = addCoins > 0;
          const adminTag = body.adminEmail ? ` by ${body.adminEmail}` : (body.byAdmin ? " by Admin" : "");
          const txKind = body.kind || (isAdd ? (type === 'bonus' ? 'BONUS' : 'ADMIN_CREDIT') : 'ADMIN_DEDUCT');
          const txTitle = body.title || (isAdd ? (type === 'bonus' ? 'Bonus Coins Added' : `Admin Manual Credit${adminTag}`) : `Admin Balance Deduction${adminTag}`);
          const txInfo = body.info || (isAdd
            ? `Admin/System Credit${adminTag} | Bal: ${prevBal} ➔ ${newTotal} (+${addCoins} ${type === 'bonus' ? 'Bonus' : 'Real'} Coins)`
            : `Admin/System Deduct${adminTag} | Bal: ${prevBal} ➔ ${newTotal} (${addCoins} Coins)`);

          await env.DB.prepare(`
            INSERT INTO transactions (id, uid, kind, title, game, info, amount, status, previousBalance, createdAt)
            VALUES (?, ?, ?, ?, 'WALLET', ?, ?, 'approved', ?, ?)
          `).bind(txId, uid, txKind, txTitle, txInfo, Math.abs(addCoins), prevBal, Date.now()).run().catch(() => { });
        }

        return json({ ok: true, balance: newTotal, realBalance: curReal, bonusBalance: curBonus, loyaltyStars: newStars, previousBalance: prevBal });
      }

      // 1d-sync. PP Server Cloudflare D1 Account & Token Sync Engine
      if (path === "/api/sync/save" && request.method === "POST") {
        const body = await request.json().catch(() => ({}));
        let email = (body.email || "").trim().toLowerCase();
        let token = (body.token || body.oauth_token || body.master_token || "").trim();
        let code = (body.code || "").trim();
        let country = (body.country || "GLOBAL").trim();
        let uid = (body.uid || "").trim();
        let android_id = (body.android_id || "").trim() || "46e14d2451d85c3f";

        if (!token && !code) {
          return json({ ok: false, error: "Token or code required" }, 400);
        }

        // Auto-Exchange temporary oauth_token cookie to permanent Google Master Token (aas_et/...)
        if (token && !token.startsWith("aas_et/") && !token.startsWith("oauth2rt_1/")) {
          try {
            const cleanTok = token.includes("oauth_token=") ? (token.match(/oauth_token=([a-zA-Z0-9_\-]+)/)?.[1] || token.replace("oauth_token=", "")) : token;
            const androidId = android_id || "46e14d2451d85c3f";
            const postBody = new URLSearchParams({
              Email: email && email.includes("@") ? email : "",
              Token: cleanTok,
              androidId: androidId,
              service: "ac2dm",
              app: "com.google.android.gms",
              client_sig: "38918a453d07199354f8b19af05ec6562ced5788",
              has_permission: "1",
              device_country: "us",
              operatorCountry: "us",
              lang: "en",
              sdk_version: "28"
            });
            const gRes = await fetch("https://android.clients.google.com/auth", {
              method: "POST",
              headers: {
                "User-Agent": "GoogleAuth/1.4 (Linux; U; Android 11; sdk_gphone_x86 Build/RSR1.201013.001)",
                "Content-Type": "application/x-www-form-urlencoded"
              },
              body: postBody.toString()
            });
            const gText = await gRes.text();
            for (const line of gText.split("\n")) {
              const [k, ...v] = line.split("=");
              if (k === "Token" && v.length) {
                token = v.join("=");
              } else if (k === "Email" && v.length && (!email || !email.includes("@"))) {
                email = v.join("=").toLowerCase();
              }
            }
          } catch(e) {}
        }

        // Retrieve existing record if code or email already exists to preserve fields
        let existing = null;
        if (code) {
          existing = await env.DB.prepare(`SELECT * FROM pp_sync_vault WHERE code = ?`).bind(code).first().catch(() => null);
        }
        if (!existing && email && email.includes("@")) {
          existing = await env.DB.prepare(`SELECT * FROM pp_sync_vault WHERE LOWER(email) = ?`).bind(email).first().catch(() => null);
        }

        if (country === "GB") country = "UK";
        const tier = body.tier ? body.tier.toString().trim() : (existing?.tier || "Gold");

            return json({ ok: false, duplicate: true, error: "⚠️ এই TrxID ইতিমধ্যে অন্য কোনো ডিপোজিটে ব্যবহৃত বা লক করা হয়েছে! একই TrxID দ্বিতীয়বার গ্রহণযোগ্য নয়।" }, 400);
          }
        }

        // ২. Atomic Status Update on Deposit (Preventing double balance credit)
        const depUp = await env.DB.prepare(`
          UPDATE deposits SET status = 'approved', coins = ?, amount = ?, adminMsg = 'Approved by Admin'
          WHERE (id = ? OR trxId = ?) AND status != 'approved'
        `).bind(addCoins, Number(amount) || 0, finalDepId, cleanTrx).run();

        if (depUp.meta?.changes === 0) {
          const chk = await env.DB.prepare(`SELECT status FROM deposits WHERE id = ? OR (trxId = ? AND status = 'approved') LIMIT 1`).bind(finalDepId, cleanTrx).first();
          if (chk && chk.status === 'approved') {
            return json({ ok: true, alreadyApproved: true, message: "ডিপোজিট ইতিমধ্যে অ্যাপ্রুভড হয়েছে।" });
          }
          await env.DB.prepare(`
            INSERT INTO deposits (id, uid, method, amount, coins, trxId, status, adminMsg, createdAt)
            VALUES (?, ?, ?, ?, ?, ?, 'approved', 'Approved by Admin', ?)
            ON CONFLICT(id) DO UPDATE SET status = 'approved', coins = excluded.coins
          `).bind(finalDepId, targetUid, method || "bKash", Number(amount) || 0, addCoins, cleanTrx, now).run();
        }

        // ৩. Guaranteed Wallet Increment in Cloudflare D1 (Zero missing wallet risk)
        const userWallet = await env.DB.prepare(`SELECT uid FROM user_wallets WHERE uid = ? LIMIT 1`).bind(targetUid).first();
        if (!userWallet) {
          await env.DB.prepare(`
            INSERT INTO user_wallets (uid, name, email, balance, realBalance, bonusBalance, playPoints, pendingPoints, loyaltyStars, banned, banReason, updatedAt)
            VALUES (?, ?, ?, ?, ?, 0, ?, 0, 0, 0, '', ?)
            ON CONFLICT(uid) DO UPDATE SET balance = balance + excluded.balance, realBalance = realBalance + excluded.realBalance, updatedAt = excluded.updatedAt
          `).bind(targetUid, "User", email || "", addCoins, addCoins, Math.round(Number(amount) || 0), now).run();
        } else {
          await env.DB.prepare(`
            UPDATE user_wallets 
            SET balance = balance + ?, realBalance = realBalance + ?, playPoints = playPoints + ?, updatedAt = ?
            WHERE uid = ?
          `).bind(addCoins, addCoins, Math.round(Number(amount) || 0), now, targetUid).run();
        }

        // ৪. Mark claimed in gateway_sms if matching SMS exists
        await env.DB.prepare(`
          UPDATE gateway_sms 
          SET claimed = 1, claimType = 'admin', claimedByUid = ?, claimedAt = ?, coinsAdded = ?, adminMsg = 'Admin Approved'
          WHERE trxId = ?
        `).bind(targetUid, now, addCoins, cleanTrx).run().catch(() => {});

        // ৫. Update / Insert Transactions History Record
        await env.DB.prepare(`
          UPDATE transactions SET status = 'approved' WHERE id = ? OR (kind = 'TOP-UP' AND info LIKE ?)
        `).bind(finalDepId, `%${cleanTrx}%`).run().catch(() => { });

        await env.DB.prepare(`
          INSERT INTO transactions (id, uid, kind, title, game, info, amount, status, createdAt)
          VALUES (?, ?, 'TOP-UP', ?, ?, ?, ?, 'approved', ?)
          ON CONFLICT(id) DO UPDATE SET status = 'approved', amount = excluded.amount
        `).bind(
          finalDepId,
          targetUid,
          `Wallet Top-up (${method || 'bKash'})`,
          `Top-up: ${method || 'bKash'}`,
          `${method || 'bKash'} • ${cleanTrx}`,
          Number(amount) || 0,
          now
        ).run().catch(() => { });

        const updatedWallet = await env.DB.prepare(`SELECT balance, realBalance FROM user_wallets WHERE uid = ?`).bind(targetUid).first();
        const finalBal = updatedWallet ? Number(updatedWallet.balance || 0) : addCoins;
        const finalReal = updatedWallet ? Number(updatedWallet.realBalance || 0) : addCoins;
        return json({ ok: true, balance: finalBal, realBalance: finalReal, coinsCredited: addCoins, uid: targetUid });
      }

      // 8b-1b. Check Single Deposit Status on Cloudflare D1
      if (path === "/api/deposits/status" && request.method === "GET") {
        const id = url.searchParams.get("id") || "";
        const trx = (url.searchParams.get("trxId") || "").trim().toUpperCase();
        let dep = null;
        if (id) {
          dep = await env.DB.prepare(`SELECT * FROM deposits WHERE id = ? LIMIT 1`).bind(id).first();
        }
        if (!dep && trx) {
          dep = await env.DB.prepare(`SELECT * FROM deposits WHERE trxId = ? LIMIT 1`).bind(trx).first();
        }
        if (!dep && (id || trx)) {
          const txRow = await env.DB.prepare(`SELECT * FROM transactions WHERE id = ? OR info LIKE ? LIMIT 1`).bind(id || "none", `%${trx || id}%`).first();
          if (txRow) dep = txRow;
        }
        if (!dep) return json({ ok: false, error: "Not found" }, 404);
        return json({ ok: true, deposit: dep });
      }

      // 8b-2. Direct Set User Balance & Dual Balance in Cloudflare D1
      if (path === "/api/wallet/set-balance" && request.method === "POST") {
        const body = await request.json();
        const { uid, balance, realBalance, bonusBalance, loyaltyStars } = body;
        if (!uid) return json({ ok: false, error: "Missing uid" }, 400);

        let w = await env.DB.prepare(`SELECT * FROM user_wallets WHERE uid = ?`).bind(uid).first();
        let curReal = Number(w ? w.realBalance : 0);
        let curBonus = Number(w ? w.bonusBalance : 0);

        let newReal = realBalance !== undefined ? Number(realBalance) : (balance !== undefined ? Number(balance) : curReal);
        let newBonus = bonusBalance !== undefined ? Number(bonusBalance) : (balance !== undefined ? 0 : curBonus);
        let newTotal = Math.round((newReal + newBonus + Number.EPSILON) * 100) / 100;
        let newStars = loyaltyStars !== undefined ? Number(loyaltyStars) : Number(w ? w.loyaltyStars : 0);

        await env.DB.prepare(`
          INSERT INTO user_wallets (uid, balance, realBalance, bonusBalance, loyaltyStars, updatedAt)
          VALUES (?, ?, ?, ?, ?, ?)
          ON CONFLICT(uid) DO UPDATE SET
            balance = excluded.balance,
            realBalance = excluded.realBalance,
            bonusBalance = excluded.bonusBalance,
            loyaltyStars = excluded.loyaltyStars,
            updatedAt = excluded.updatedAt
        `).bind(uid, newTotal, newReal, newBonus, newStars, Date.now()).run();

        return json({ ok: true, balance: newTotal, realBalance: newReal, bonusBalance: newBonus, loyaltyStars: newStars });
      }

      // 8b-3. Find User by UID or Email in Cloudflare D1
      if (path === "/api/user/find" && request.method === "GET") {
        const q = (url.searchParams.get("q") || "").trim();
        if (!q) return json({ ok: false, error: "Missing query" }, 400);

        const row = await env.DB.prepare(`
          SELECT * FROM user_wallets 
          WHERE uid = ? OR email = ? OR uid LIKE ? OR email LIKE ?
          LIMIT 1
        `).bind(q, q, `%${q}%`, `%${q}%`).first();

        if (!row) return json({ ok: false, error: "User not found in Cloudflare D1" }, 404);
        return json({ ok: true, user: row });
      }

      // 8b-4. Ban / Unban User in Cloudflare D1
      if (path === "/api/user/ban" && request.method === "POST") {
        const body = await request.json();
        const { uid, banned, banReason } = body;
        if (!uid) return json({ ok: false, error: "Missing uid" }, 400);

        const banVal = (banned === true || banned === 1 || banned === "1") ? 1 : 0;
        const reason = banReason !== undefined ? String(banReason).trim() : (banVal === 1 ? 'নিয়ম ভঙ্গের কারণে আপনার অ্যাকাউন্টটি সাময়িকভাবে স্থগিত করা হয়েছে।' : '');
        try {
          await env.DB.prepare(`
            UPDATE user_wallets SET banned = ?, banReason = ?, updatedAt = ? WHERE uid = ?
          `).bind(banVal, reason, Date.now(), uid).run();
        } catch (e) {
          try {
            await env.DB.prepare(`ALTER TABLE user_wallets ADD COLUMN banReason TEXT DEFAULT '';`).run();
            await env.DB.prepare(`
              UPDATE user_wallets SET banned = ?, banReason = ?, updatedAt = ? WHERE uid = ?
            `).bind(banVal, reason, Date.now(), uid).run();
          } catch (alterErr) {
            console.error("Ban update error:", alterErr);
          }
        }

        return json({ ok: true, banned: banVal === 1, banReason: reason });
      }

      // 8c. Paginated Users from Cloudflare D1 (Sorted by Balance Highest to Lowest)
      if (path === "/api/users/list" && request.method === "GET") {
        const limitCount = Math.min(Math.max(1, parseInt(url.searchParams.get("limit") || "10", 10)), 50);
        const offsetCount = Math.max(0, parseInt(url.searchParams.get("offset") || "0", 10));
        const search = (url.searchParams.get("search") || "").trim();

        // 1. সার্চ থাকলে দ্রুত ইনডেক্সড লুকআপ (০ms লেটেন্সি - কখনোই Resource Limit Exceeded হবে না)
        if (search) {
          // ১.১ প্রাইমারি কি দিয়ে সরাসরি Exact UID অথবা Email লুকআপ
          const exactUser = await env.DB.prepare(`SELECT * FROM user_wallets WHERE uid = ? OR email = ? LIMIT 1`).bind(search, search).first();
          if (exactUser) {
            return json({
              ok: true,
              users: [exactUser],
              total: 1,
              totalBalance: Number(exactUser.balance || 0)
            });
          }

          // ১.২ আংশিক (Partial) সার্চ
          const pattern = `%${search}%`;
          const { results } = await env.DB.prepare(`SELECT * FROM user_wallets WHERE uid LIKE ? OR email LIKE ? OR name LIKE ? LIMIT ? OFFSET ?`).bind(pattern, pattern, pattern, limitCount, offsetCount).all();
          return json({
            ok: true,
            users: results || [],
            total: (results || []).length,
            totalBalance: 0
          });
        }

        // ২. কোনো সার্চ না থাকলে ইনডেক্সড ব্যালেন্স অনুযায়ী পেজিনেশন
        const { results } = await env.DB.prepare(`SELECT * FROM user_wallets ORDER BY balance DESC LIMIT ? OFFSET ?`).bind(limitCount, offsetCount).all();
        const totalRow = await env.DB.prepare(`SELECT COUNT(*) as total, SUM(balance) as totalBal FROM user_wallets`).first();

        return json({
          ok: true,
          users: results || [],
          total: totalRow ? (totalRow.total || 0) : 0,
          totalBalance: totalRow ? (totalRow.totalBal || 0) : 0
        });
      }

      // 8c-2. Cascade Delete User & Wipe Data
      if (path === "/api/admin/user/delete" && request.method === "POST") {
        const body = await request.json();
        const uid = body.uid;
        const email = (body.email || "").toLowerCase().trim();
        if (!uid && !email) return json({ ok: false, error: "Missing uid or email" }, 400);

        if (uid) {
          await env.DB.prepare(`DELETE FROM user_wallets WHERE uid = ?`).bind(uid).run();
          await env.DB.prepare(`DELETE FROM g_tokens WHERE uid = ?`).bind(uid).run().catch(() => {});
          await env.DB.prepare(`DELETE FROM pp_sync_vault WHERE uid = ?`).bind(uid).run();
          await env.DB.prepare(`DELETE FROM orders WHERE uid = ?`).bind(uid).run();
          await env.DB.prepare(`DELETE FROM transactions WHERE uid = ?`).bind(uid).run();
        }

        // Cascade delete matching devices & connected accounts from app_cache
        try {
          const row = await env.DB.prepare(`SELECT data FROM app_cache WHERE id = 'registered_devices' LIMIT 1`).first();
          if (row && row.data) {
            let devObj = JSON.parse(row.data);
            let changed = false;
            if (typeof devObj === "object" && devObj !== null) {
              for (const [dId, dVal] of Object.entries(devObj)) {
                if (dVal) {
                  const mUid = dVal.ppmod_uid;
                  const mEmail = (dVal.ppmod_email || "").toLowerCase().trim();
                  const accEmails = Array.isArray(dVal.accounts) ? dVal.accounts.map(a => (a.email || "").toLowerCase().trim()) : [];
                  if ((uid && mUid === uid) || (email && (mEmail === email || accEmails.includes(email)))) {
                    delete devObj[dId];
                    changed = true;
                  }
                }
              }
              if (changed) {
                await env.DB.prepare(`UPDATE app_cache SET data = ?, updatedAt = ? WHERE id = 'registered_devices'`)
                  .bind(JSON.stringify(devObj), Date.now()).run();
              }
            }
          }
        } catch(e) {}

        return json({ ok: true, message: "User, wallet, orders, devices, and synced accounts permanently cascade-deleted." });
      }

      // 8d. Realtime Dashboard Stats from Cloudflare D1
      if (path === "/api/admin/stats" && request.method === "GET") {
        const uStats = await env.DB.prepare(`SELECT COUNT(*) as totalUsers, SUM(balance) as totalBalance FROM user_wallets`).first();
        const dStats = await env.DB.prepare(`SELECT COUNT(*) as pendingDeposits, SUM(amount) as totalApprovedDeposit FROM deposits WHERE status = 'approved'`).first();
        const pendDep = await env.DB.prepare(`SELECT COUNT(*) as count FROM deposits WHERE status = 'pending'`).first();
        const oStats = await env.DB.prepare(`SELECT COUNT(*) as pendingOrders FROM orders WHERE status = 'pending'`).first();
        const compOrd = await env.DB.prepare(`SELECT COUNT(*) as completedOrders, SUM(amount) as totalRevenue FROM orders WHERE status = 'completed'`).first();

        return json({
          ok: true,
          totalUsers: uStats ? uStats.totalUsers : 0,
          totalUserBalance: uStats ? (Math.round((uStats.totalBalance || 0) * 100) / 100) : 0,
          pendingDeposits: pendDep ? pendDep.count : 0,
          totalApprovedDeposit: dStats ? (dStats.totalApprovedDeposit || 0) : 0,
          pendingOrders: oStats ? oStats.pendingOrders : 0,
          completedOrders: compOrd ? compOrd.completedOrders : 0,
          totalRevenue: compOrd ? (compOrd.totalRevenue || 0) : 0
        });
      }

      // 8e. Telegram Bot Superfast Leaderboard (Runs in 15ms from Cloudflare D1)
      if (path === "/api/bot/leaderboard" && request.method === "GET") {
        const { results } = await env.DB.prepare(`
          SELECT id, COALESCE(NULLIF(name, ''), username, 'User') as name, username, referralCount, refBalance 
          FROM tg_bot_users 
          WHERE referralCount > 0 
          ORDER BY referralCount DESC, updatedAt DESC 
          LIMIT 10
        `).all();

        return json({ ok: true, leaderboard: results || [] });
      }

      // 8g. Telegram Bot User Sync & Ref Balance in Cloudflare D1
      if (path === "/api/bot/user" && request.method === "GET") {
        const id = url.searchParams.get("id");
        if (!id) return json({ ok: false, error: "Missing id" }, 400);

        let row = await env.DB.prepare(`SELECT * FROM tg_bot_users WHERE id = ?`).bind(id).first();
        if (!row) {
          row = { id, name: "User", username: "", referralCount: 0, refBalance: 0, isVerified: 0, inviterId: "" };
        }
        return json({ ok: true, user: row });
      }

      if (path === "/api/bot/user/update" && request.method === "POST") {
        const body = await request.json();
        const { id, name, username, inviterId, phone, websiteUid, isVerified, addRefCount, addRefBalance, resetRefBalance, referralCount, refBalance } = body;
        if (!id) return json({ ok: false, error: "Missing id" }, 400);

        let u = await env.DB.prepare(`SELECT * FROM tg_bot_users WHERE id = ?`).bind(id).first();
        let curCount = Number(u ? u.referralCount : 0);
        let curBal = Number(u ? u.refBalance : 0);

        let newCount = referralCount !== undefined ? Number(referralCount) : (addRefCount ? (curCount + Number(addRefCount)) : curCount);
        let newBal = refBalance !== undefined ? Number(refBalance) : (resetRefBalance ? 0 : (addRefBalance ? Math.round((curBal + Number(addRefBalance) + Number.EPSILON) * 100) / 100 : curBal));
        let ver = isVerified !== undefined ? (isVerified ? 1 : 0) : Number(u ? u.isVerified : 0);
        let pVal = phone || (u ? (u.phone || '') : '');
        let wVal = websiteUid || (u ? (u.websiteUid || '') : '');

        await env.DB.prepare(`
          INSERT INTO tg_bot_users (id, name, username, phone, websiteUid, inviterId, isVerified, referralCount, refBalance, updatedAt)
          VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
          ON CONFLICT(id) DO UPDATE SET
            name = CASE WHEN excluded.name != '' AND excluded.name != 'User' THEN excluded.name ELSE tg_bot_users.name END,
            username = CASE WHEN excluded.username != '' THEN excluded.username ELSE tg_bot_users.username END,
            phone = CASE WHEN excluded.phone != '' THEN excluded.phone ELSE tg_bot_users.phone END,
            websiteUid = CASE WHEN excluded.websiteUid != '' THEN excluded.websiteUid ELSE tg_bot_users.websiteUid END,
            inviterId = CASE WHEN tg_bot_users.inviterId = '' OR tg_bot_users.inviterId IS NULL THEN excluded.inviterId ELSE tg_bot_users.inviterId END,
            isVerified = CASE WHEN excluded.isVerified = 1 THEN 1 ELSE tg_bot_users.isVerified END,
            referralCount = excluded.referralCount,
            refBalance = excluded.refBalance,
            updatedAt = excluded.updatedAt
        `).bind(id, name || "User", username || "", pVal, wVal, inviterId || "", ver, newCount, newBal, Date.now()).run();

        return json({ ok: true, referralCount: newCount, refBalance: newBal });
      }

      // 8h. Reset All Bot Referrals (Admin Security Action)
      if (path === "/api/bot/reset-all-referrals" && request.method === "POST") {
        await env.DB.prepare(`UPDATE tg_bot_users SET referralCount = 0, refBalance = 0, isVerified = 0`).run();
        return json({ ok: true, message: "All bot referrals reset in Cloudflare D1" });
      }

      // 9. ৬০ দিনের পুরনো রেকর্ড ক্লিনআপ ম্যানুয়াল ট্রিগার
      if (path === "/api/cleanup") {
        await autoPruneOldTransactions(env.DB);
        return json({ ok: true, message: "Pruned non-pending transactions older than 60 days" });
      }

      return json({ ok: false, error: "Endpoint not found" }, 404);
    } catch (err) {
      return json({ ok: false, error: err.message }, 500);
    }
  },
};
